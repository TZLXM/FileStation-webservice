## 1. DTO 类全集

所有 DTO 使用 class-validator 装饰器。ValidationPipe 必须开启 `transform: true`（配合 `@Type`），`whitelist: true, forbidNonWhitelisted: true` 才能作用于 class。

`apps/server/src/uploads/dto/init-upload.dto.ts`:
```typescript
import { IsString, Length, IsInt, Min, Max, IsOptional, Matches } from 'class-validator';
import { Type } from 'class-transformer';

export const MAX_FILE_SIZE = 100 * 1024 * 1024 * 1024; // 100 GiB 上限（可调）
export const MIN_CHUNK_SIZE = 64 * 1024;   // 64 KiB
export const MAX_CHUNK_SIZE = 64 * 1024 * 1024; // 64 MiB
export const SHA256_HEX = /^[a-f0-9]{64}$/;

export class InitUploadDto {
  @IsString()
  @Length(1, 255)
  filename!: string;

  @Type(() => Number)
  @IsInt()
  @Min(0)
  @Max(MAX_FILE_SIZE)
  size!: number;

  @IsOptional()
  @IsString()
  @Matches(SHA256_HEX, { message: 'hash must be a lowercase sha256 hex digest' })
  hash?: string;

  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(MIN_CHUNK_SIZE)
  @Max(MAX_CHUNK_SIZE)
  chunk_size?: number;
}
```

`apps/server/src/files/dto/update-file.dto.ts` — **null 语义**：字段缺失 = 不修改；显式 `null` = 永久保留。用 `@ValidateIf` 区分：
```typescript
import { IsString, Length, IsOptional, IsUUID, IsISO8601, ValidateIf } from 'class-validator';

export class UpdateFileDto {
  @IsOptional()
  @IsString()
  @Length(1, 255)
  filename?: string;

  // undefined = 不移动；null = 移回收件箱（根级）；string = 目标文件夹
  @ValidateIf((_obj, value) => value !== null && value !== undefined)
  @IsUUID()
  folder_id?: string | null;

  // undefined = 不修改过期时间；null = 永久保留；string = 新的 ISO 过期时间
  @ValidateIf((_obj, value) => value !== null && value !== undefined)
  @IsISO8601()
  expires_at?: string | null;
}
```
注意：不要给 `folder_id` / `expires_at` 再加 `@IsOptional()`——`@IsOptional` 会同时跳过 `null` 和 `undefined` 的校验，导致 `folder_id: "not-a-uuid"` 之外的非法值漏网；`@ValidateIf` 已精确表达三种状态。配合 `whitelist: true`（不会剥离已声明字段）与 `forbidNonWhitelisted: true`（拒绝未知字段）。

### 2.1 合并 access/verify 为单端点

`apps/server/src/shares/shares.service.ts` — 新增 `accessShare`，**删除** `createAccess` 与 `verifyPassword`（以及 SharesController 中 `@Post(':id/verify')` 端点）。bcrypt.compare 在任何 DB 写之前完成：

```typescript
async accessShare(
  shareId: string,
  password?: string,
): Promise<{ download_token: string; expires_at: string; file_info: ShareInfo }> {
  // getShareInfo 统一校验：存在/未过期/未吊销/未耗尽/文件 active
  const shareInfo = await this.getShareInfo(shareId);

  if (shareInfo.protection === ShareProtection.PASSWORD) {
    // 必须携带密码
    if (!password) {
      throw new BadRequestException({
        code: 'PASSWORD_REQUIRED',
        message: 'This share requires a password',
      });
    }
    const share = await this.sharesRepository.findOne({ where: { id: shareId } });
    if (!share || !share.passwordHash) {
      throw new BadRequestException({ code: 'INVALID_SHARE', message: 'Share password not set' });
    }
    const isValid = await bcrypt.compare(password, share.passwordHash);
    if (!isValid) {
      throw new BadRequestException({ code: 'INVALID_PASSWORD', message: 'Invalid password' });
    }
  }
  // protection === 'none'：忽略 body 中的 password，直接放行

  const downloadToken = uuidv4();
  const expiresAt = Date.now() + 60 * 60 * 1000; // 1 小时

  const session = this.downloadSessionsRepository.create({
    id: uuidv4(),
    shareId,
    fileId: shareInfo.file_id,
    tokenHash: this.hashToken(downloadToken),
    counted: 0,
    createdAt: Date.now(),
    expiresAt,
  });
  await this.downloadSessionsRepository.save(session);

  return {
    download_token: downloadToken,
    expires_at: new Date(expiresAt).toISOString(),
    file_info: shareInfo,
  };
}
```

### 2.6 FilesController PATCH 支持 expires_at: null

`apps/server/src/files/files.controller.ts`：
```typescript
@Patch(':id')
async update(
  @Param('id') id: string,
  @Body() dto: UpdateFileDto,
): Promise<ApiResponse<FileMetadata>> {
  const updateData: Partial<File> = {};
  if (dto.filename !== undefined) updateData.filename = dto.filename;

  if (dto.folder_id !== undefined) {
    // null = 收件箱；string 时 FilesService.update 内校验目标文件夹存在且未删除（问题 9 配合修正）
    updateData.folderId = dto.folder_id;
  }

  if (dto.expires_at !== undefined) {
    if (dto.expires_at === null) {
      // 显式 null = 永久保留：清除过期与已过期标记，恢复 active
      updateData.expiresAt = null;
      updateData.expiredAt = null;
      updateData.status = FileStatus.ACTIVE;
    } else {
      const ts = new Date(dto.expires_at).getTime();
      updateData.expiresAt = ts;
      // 若新过期时间在未来，同时清除过期标记
      if (ts > Date.now()) {
        updateData.expiredAt = null;
        updateData.status = FileStatus.ACTIVE;
      }
    }
  }

  const file = await this.filesService.update(id, updateData);
  return { code: 'OK', message: 'File updated', data: this.toMetadata(file), request_id: crypto.randomUUID() };
}
```

## 3. 初始化部署与启动顺序

### 3.1 main.ts 重写（修复启动竞态：token 生成在 listen 之前）

`apps/server/src/main.ts`:
```typescript
import { NestFactory } from '@nestjs/core';
import { ValidationPipe, Logger } from '@nestjs/common';
import { AppModule } from './app.module';
import { ConfigService } from '@nestjs/config';
import { AuthService } from './auth/auth.service';
import { StorageService } from './files/storage.service';
import { UploadsService } from './files/uploads.service';
import cookieParser from 'cookie-parser';
import * as fs from 'fs/promises';
import { dirname } from 'path';

async function bootstrap() {
  const logger = new Logger('Bootstrap');

  // ========== 阶段 1：目录预创建（NestFactory.create 之前，纯配置读取，无 DI） ==========
  // DatabaseModule 连接 SQLite 前 db 父目录必须存在，否则 better-sqlite3/sqlite3 报 SQLITE_CANTOPEN
  const storagePath = process.env.FILESTATION_STORAGE_PATH || './data/storage';
  const tempPath = process.env.FILESTATION_TEMP_PATH || './data/temp';
  const dbPath = process.env.FILESTATION_DB_PATH || './data/filestation.db';
  await fs.mkdir(storagePath, { recursive: true });
  await fs.mkdir(tempPath, { recursive: true });
  await fs.mkdir(dirname(dbPath), { recursive: true });
  logger.log(`Directories ensured: storage=${storagePath}, temp=${tempPath}, db=${dirname(dbPath)}`);

  // ========== 阶段 2：创建应用（migrationsRun 在此触发） ==========
  const app = await NestFactory.create(AppModule);
  const configService = app.get(ConfigService);

  app.use(cookieParser());
  app.setGlobalPrefix('api/v1'); // 唯一前缀来源，Controller 无前缀
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      transform: true,
      forbidNonWhitelisted: true,
      transformOptions: { enableImplicitConversion: false },
    }),
  );
  app.enableCors({
    origin: ['http://localhost:5173', 'http://127.0.0.1:5173'],
    credentials: true,
  });

  // ========== 阶段 3：app.init() 触发 OnApplicationBootstrap ==========
  // DatabaseInitializer（PRAGMA + foreign_keys 校验）在此执行，migrationsRun 已完成
  await app.init();

  // StorageService 构造仅解析路径；目录已在阶段 1 创建，此处幂等再确保（含 ConfigService 实际生效路径）
  const storageService = app.get(StorageService);
  await storageService.ensureDirectories();

  // ========== 阶段 4：init token 覆盖 —— 必须在 listen 之前 ==========
  // 修复 v1.5 竞态：旧 token 在 listen 后才被覆盖，存在窗口期。
  // 现在在端口开放前完成覆盖，监听后只有新 token 有效。
  const authService = app.get(AuthService);
  const initToken = await authService.ensureInitToken();
  if (initToken) {
    logger.log('================================');
    logger.log('FileStation 首次启动（未初始化）');
    logger.log(`初始化 Token: ${initToken}`);
    logger.log('有效期：10 分钟，仅本机可访问 /api/v1/auth/init');
    logger.log('请通过本机访问 http://localhost/init 完成设置');
    logger.log('================================');
  }

  // ========== 阶段 5：恢复扫描（listen 之前，无文件 I/O 冲突） ==========
  // 上次进程退出遗留的 verifying/uploading 会话恢复、孤儿临时目录清理
  const uploadsService = app.get(UploadsService);
  await uploadsService.recoverOnStartup();

  // ========== 阶段 6：最后才监听 ==========
  const port = configService.get<number>('app.port') || 8080;
  await app.listen(port, '127.0.0.1');

  logger.log(`FileStation server running on http://127.0.0.1:${port}`);
  logger.log(`Environment: ${configService.get<string>('app.nodeEnv')}`);
}

bootstrap();
```

`apps/server/src/files/storage.service.ts` — `ensureDirectories` 补 `OnApplicationBootstrap` 接口（双保险，main.ts 阶段 4 已显式调用；接口实现保证在其它启动路径如 E2E `app.init()` 下也生效）：

### 4.4 FileLifecycleService 消费 cleanup_grace_hours

`apps/server/src/files/file-lifecycle.service.ts`（v1.5 仅有文件占位，v1.6 给出实现）：
```typescript
import { Injectable, Logger } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, LessThan, In } from 'typeorm';
import { File, FileStatus } from './entities/file.entity';
import { StorageService } from './storage.service';
import { SettingsService } from '../settings/settings.service';

@Injectable()
export class FileLifecycleService {
  private readonly logger = new Logger(FileLifecycleService.name);

  constructor(
    @InjectRepository(File)
    private filesRepository: Repository<File>,
    private storageService: StorageService,
    private settingsService: SettingsService,
  ) {}

  @Cron(CronExpression.EVERY_10_MINUTES)
  async processLifecycle(): Promise<void> {
    const now = Date.now();

    // 阶段 A：标记过期（expiresAt 非 null 且已过期的 active 文件）
    await this.filesRepository
      .createQueryBuilder()
      .update(File)
      .set({ status: FileStatus.EXPIRED, expiredAt: now, updatedAt: now })
      .where('status = :active', { active: FileStatus.ACTIVE })
      .andWhere('expires_at IS NOT NULL')
      .andWhere('expires_at < :now', { now })
      .execute();

    // 阶段 B：宽限期后物理删除（消费 cleanup_grace_hours）
    const { cleanup_grace_hours } = await this.settingsService.getStorageSettings();
    const graceCutoff = now - cleanup_grace_hours * 60 * 60 * 1000;

    // 过期且超过宽限期 的 + 标记 deleting 且超过宽限期 的文件
    const doomed = await this.filesRepository.find({
      where: [
        { status: FileStatus.EXPIRED, expiredAt: LessThan(graceCutoff) },
        { status: FileStatus.DELETING, updatedAt: LessThan(graceCutoff) },
      ],
      take: 100, // 单批上限，防长事务/长循环
    });

    for (const file of doomed) {
      // 文件 I/O 在事务外：先删磁盘，后更新 DB。
      // 若磁盘删除失败，保留 deleting/expired 状态，下轮重试。
      try {
        await this.storageService.deleteFile(file.storedName);
      } catch (err) {
        this.logger.warn(`Failed to delete stored file ${file.storedName}: ${err}`);
        continue;
      }
      await this.filesRepository.update(file.id, {
        status: FileStatus.DELETED,
        deletedAt: now,
        updatedAt: Date.now(),
      });
    }
  }
}
```
（需 File 实体有 `deletedAt` 字段；若 v1.5 实体缺，迁移中补 `deleted_at INTEGER`。`files.delete()` 置 DELETING 后由本任务宽限后真正清理——修复"无人清理"。）

### 4.5 登录锁定：消费 max_login_attempts / lockout_minutes

轻量实现：复用 `system_meta` 表（key=`login_lockout`），不新建表。失败计数与锁定状态为单 JSON 值；校验/重置用条件 UPDATE 避免并发计数丢失（SQLite 不支持 FOR UPDATE）。

`apps/server/src/auth/auth.service.ts`：
```typescript
import { SettingsService } from '../settings/settings.service';

// AuthService 构造函数新增：
//   private settingsService: SettingsService,
// （AuthModule imports 增加 SettingsModule；SettingsModule 不依赖 AuthModule 的 provider 运行期，
//  仅 import JwtAuthGuard——若构成循环，将 JwtAuthGuard 移至共享 security 模块或用 forwardRef。
//  推荐：SettingsModule 不 import AuthModule，SettingsController 的 guard 改用全局 APP_GUARD
//  或在 SettingsModule 中 imports: [forwardRef(() => AuthModule)]。）

private static readonly LOCKOUT_KEY = 'login_lockout';

interface LockoutState {
  failed_count: number;
  locked_until: number | null; // Unix ms
}

private async getLockoutState(): Promise<LockoutState> {
  const meta = await this.systemMetaRepository.findOne({ where: { key: AuthService.LOCKOUT_KEY } });
  if (!meta) return { failed_count: 0, locked_until: null };
  try {
    return JSON.parse(meta.value) as LockoutState;
  } catch {
    return { failed_count: 0, locked_until: null };
  }
}

private async setLockoutState(state: LockoutState): Promise<void> {
  await this.systemMetaRepository.save({
    key: AuthService.LOCKOUT_KEY,
    value: JSON.stringify(state),
  });
}

async login(loginDto: LoginDto): Promise<TokenPair> {
  const security = await this.settingsService.getSecuritySettings();
  const now = Date.now();

  // 1) 锁定检查（在任何凭证验证之前，防锁定期间探测密码）
  const state = await this.getLockoutState();
  if (state.locked_until && state.locked_until > now) {
    const retryAfterSec = Math.ceil((state.locked_until - now) / 1000);
    throw new UnauthorizedException({
      code: 'ACCOUNT_LOCKED',
      message: `Account locked, retry after ${retryAfterSec} seconds`,
      retry_after: retryAfterSec,
    });
  }

  const account = await this.accountsService.findByUsername(loginDto.username);
  // 用户不存在与密码错误走同一路径，防用户名枚举
  const isValid = account
    ? await this.accountsService.validatePassword(account, loginDto.password)
    : (await import('bcrypt').then(b => b.hash(loginDto.password, 10)).then(() => false), false);

  if (!account || !isValid) {
    // 2) 失败计数（宽限窗口重置：上次锁定已过期则从 1 重新计）
    const base = state.locked_until && state.locked_until <= now ? 0 : state.failed_count;
    const failedCount = base + 1;
    const newState: LockoutState = { failed_count: failedCount, locked_until: null };
    if (failedCount >= security.max_login_attempts) {
      newState.locked_until = now + security.lockout_minutes * 60 * 1000;
      newState.failed_count = 0; // 锁定后清零，解锁后重新计
    }
    await this.setLockoutState(newState);
    throw new UnauthorizedException('Invalid credentials');
  }

  // 3) 成功：重置计数
  if (state.failed_count > 0 || state.locked_until) {
    await this.setLockoutState({ failed_count: 0, locked_until: null });
  }

  return this.generateTokens(account.id, account.username);
}
```
（说明：失败时的 `await import('bcrypt').then(b => b.hash(...))` 仅为拉平计时侧信道，可选；锁定的安全性不依赖它。system_meta 单 key 读写并发竞争在 SQLite busy_timeout 下安全——最坏情况是两次并发失败只记一次，可接受的宽松计数。）

### 4.6 totp_required 决策

**决策：双层拒绝（DTO 层 + 前端隐藏），不静默忽略。**

- 后端：`SecuritySettingsDto.totp_required` 用 `@IsIn([false])` 装饰——传 `true` 直接 400，传 `false` 等价于无操作。理由：静默忽略会让管理员误以为已开启 TOTP（安全功能假开启比不存在更危险）；显式 400 把"Phase 1 不支持"变成可见错误。
- 前端：`apps/web/src/pages/SettingsPage.tsx` 安全设置区块不渲染 `totp_required` 字段（GET 返回中该字段固定 false，仅忽略不展示）：
```tsx
{/* Phase 1 不渲染 totp_required；后端 DTO 层亦拒绝写入 true */}
<div className="space-y-4">
  <NumberField label="最大登录失败次数" value={settings.security.max_login_attempts}
    onChange={(v) => updateSecurity('max_login_attempts', v)} />
  <NumberField label="锁定时长（分钟）" value={settings.security.lockout_minutes}
    onChange={(v) => updateSecurity('lockout_minutes', v)} />
```
