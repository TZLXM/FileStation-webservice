## 2. StorageService 改动（替换 v1.5 中 writePart/deletePart）

`getPartPath` / `getPartTempPath` 签名与路径格式**保持不变**（`temp/<upload_id>/part_NNNNNN.part` 与 `part_NNNNNN.<uniqueId>.part.tmp`）。拆分 v1.5 一体化的 `writePart` 为三个方法，使事务内永远不需要调用 StorageService。

`apps/server/src/files/storage.service.ts`（仅列出变更方法，其余方法与 v1.5 相同）：

```typescript
  /**
   * 写 owner 专属临时文件（write + fsync），并校验大小与 checksum。
   * 不做 rename，不触碰正式分块路径。校验失败只删除该临时文件。
   * 仅在事务外调用。
   */
  async writePartToTemp(
    uploadId: string,
    partNumber: number,
    uniqueId: string,
    data: Buffer,
    expectedChecksum: string,
  ): Promise<void> {
    const tempPath = this.getPartTempPath(uploadId, partNumber, uniqueId);

    const handle = await fs.open(tempPath, 'w');
    try {
      await handle.writeFile(data);
      await handle.sync(); // 真实 fsync，保证 rename 前数据落盘
    } finally {
      await handle.close();
    }

    const actualChecksum = createHash('sha256').update(data).digest('hex');
    if (actualChecksum !== expectedChecksum) {
      await fs.unlink(tempPath).catch(() => {});
      throw new BadRequestException({
        code: 'CHECKSUM_MISMATCH',
        message: 'Part checksum mismatch',
        expected: expectedChecksum,
        actual: actualChecksum,
      });
    }
  }

  /**
   * 把 owner 的临时文件原子 rename 到正式分块路径。
   * 仅在事务外调用。失败后调用方负责 deleteTempPart。
   */
  async renamePart(uploadId: string, partNumber: number, uniqueId: string): Promise<void> {
    const tempPath = this.getPartTempPath(uploadId, partNumber, uniqueId);
    const finalPath = this.getPartPath(uploadId, partNumber);
    await fs.rename(tempPath, finalPath);
  }

  /**
   * 删除 owner 自己的临时文件（只删带 uniqueId 的临时路径，绝不删正式分块）。
   * 幂等：文件不存在时静默成功。
   */
  async deleteTempPart(uploadId: string, partNumber: number, uniqueId: string): Promise<void> {
    const tempPath = this.getPartTempPath(uploadId, partNumber, uniqueId);
    await fs.unlink(tempPath).catch(() => {});
  }

  /**
   * 删除正式分块路径（仅供 abort/cleanup 使用，禁止在事务内或 uploadPart 正常路径调用）。
   */
  async deletePart(uploadId: string, partNumber: number): Promise<void> {
    const path = this.getPartPath(uploadId, partNumber);
    await fs.unlink(path).catch(() => {});
  }
```

注：v1.5 的 `writePart()` 删除。`uploadPart` 正常路径**不再存在"删除正式分块"**——owner 失败只删自己的 `part_NNNNNN.<uniqueId>.part.tmp`，公共 `part_NNNNNN.part` 只在 DB 行从 `receiving`→`ready` 之后才会被新的 ready owner 覆盖 rename（见下文阶段三说明）。

## 3. uploadPart() 重写 —— 三阶段 owner 抢占（替换 v1.5 第 2662–2761 行）

设计要点：

- 阶段一短事务（`BEGIN IMMEDIATE` + 原生 SQL）：`INSERT ... SELECT WHERE NOT EXISTS(ready)` 抢占，或行已存在时按 `status`/`checksum` 分支。事务内**零文件 I/O**。
- 阶段二事务外：写临时文件 + fsync + 校验 + rename。
- 阶段三短事务：条件 UPDATE `receiving→ready`（带 owner_token 与 status 守卫），仅 `affected=1` 时 increment `received_size`。
- owner 阶段二崩溃：DB 行停在 `receiving`，临时文件为孤儿 → 由第 5 节 `recoverStaleReceivingParts` / `scanOrphanTempFiles` 回收。

`apps/server/src/files/uploads.service.ts`（替换 uploadPart，新增辅助方法）：

```typescript
  /**
   * 上传分块：三阶段 owner 抢占协议。
   * 阶段一：短事务抢占/幂等判定（无文件 I/O）
   * 阶段二：事务外写临时文件 -> fsync -> 校验 -> rename 到正式路径
   * 阶段三：短事务 receiving -> ready + 首次递增 received_size
   */
  async uploadPart(
    uploadId: string,
    partNumber: number,
    data: Buffer,
    checksum: string,
    uploadToken: string,
  ): Promise<{ received: boolean; part_checksum: string }> {
    // Token + 状态校验（事务外读，阶段一内还有原子守卫，此处仅快速失败）
    const session = await this.validateUploadToken(uploadId, uploadToken, [
      UploadStatus.INITIATED,
      UploadStatus.UPLOADING,
    ]);

    // 分块编号校验
    const totalParts = Math.ceil(session.expectedSize / session.chunkSize);
    if (!Number.isInteger(partNumber) || partNumber < 0 || partNumber >= totalParts) {
      throw new BadRequestException({
        code: 'INVALID_PART_NUMBER',
        message: `Part number must be between 0 and ${totalParts - 1}`,
      });
    }

    // 分块大小校验
    const expectedSize = partNumber === totalParts - 1
      ? session.expectedSize - partNumber * session.chunkSize
      : session.chunkSize;
    if (data.length !== expectedSize) {
      throw new BadRequestException({
        code: 'INVALID_PART_SIZE',
        message: `Part size must be ${expectedSize} bytes`,
        expected: expectedSize,
        actual: data.length,
      });
    }

    const ownerToken = uuidv4();          // 本请求的 owner 令牌
    const tempName = `${partNumber.toString().padStart(6, '0')}.${ownerToken}`; // 临时文件唯一名（不含扩展，仅 DB 记录用）
    const offset = partNumber * session.chunkSize;
    const now = Date.now();

    // ---------- 阶段一：短事务抢占（BEGIN IMMEDIATE + 原生 SQL，无文件 I/O）----------
    const claim = await this.claimPart(uploadId, partNumber, {
      offset,
      size: data.length,
      checksum,
      ownerToken,
      tempName,
      now,
    });

    if (claim.outcome === 'already_ready') {
      // 幂等：已 ready 且 checksum 一致
      return { received: true, part_checksum: checksum };
    }
    // outcome === 'claimed'：本请求获得接收权，进入阶段二

    // ---------- 阶段二：事务外文件 I/O ----------
    try {
      await this.storageService.writePartToTemp(uploadId, partNumber, ownerToken, data, checksum);
      await this.storageService.renamePart(uploadId, partNumber, ownerToken);
    } catch (error) {
      // 只删除自己的临时文件，绝不触碰正式分块路径
      await this.storageService.deleteTempPart(uploadId, partNumber, ownerToken);

      if (error instanceof BadRequestException) {
        // checksum 不匹配：数据本身错误，释放接收权让后续重传可以抢占
        await this.releasePartClaim(uploadId, partNumber, ownerToken);
      }
      // 其他 I/O 错误（磁盘满等）：保留 receiving 行，
      // 由 recoverStaleReceivingParts() 超时回收，防止错误数据被误认为已接收
      throw error;
    }

    // ---------- 阶段三：短事务确认 receiving -> ready ----------
    await this.confirmPartReady(uploadId, partNumber, ownerToken, data.length);

    return { received: true, part_checksum: checksum };
  }

  /**
   * 阶段一：在 BEGIN IMMEDIATE 事务内抢占分块接收权。
   * 返回 'already_ready'（幂等命中）或 'claimed'（获得接收权）。
   * 冲突抛 409 / 400。
   */
  private async claimPart(
    uploadId: string,
    partNumber: number,
    info: { offset: number; size: number; checksum: string; ownerToken: string; tempName: string; now: number },
  ): Promise<{ outcome: 'already_ready' | 'claimed' }> {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    try {
      await queryRunner.query('BEGIN IMMEDIATE');

      // 1) 原子守卫：会话仍处于可上传状态（防止 complete 抢占后仍有分块写入）
      const sessionRows = await queryRunner.query(
        `SELECT status FROM upload_sessions WHERE id = ?`,
        [uploadId],
      );
      if (sessionRows.length === 0) {
        throw new NotFoundException('Upload session not found');
      }
      if (sessionRows[0].status !== UploadStatus.INITIATED && sessionRows[0].status !== UploadStatus.UPLOADING) {
        throw new BadRequestException({
          code: 'INVALID_UPLOAD_STATE',
          message: `Operation not allowed in state: ${sessionRows[0].status}`,
        });
      }

      // 2) 已存在且 ready：按 checksum 判定幂等或冲突
      const existing = await queryRunner.query(
        `SELECT status, checksum, owner_token FROM upload_parts WHERE upload_id = ? AND part_number = ?`,
        [uploadId, partNumber],
      );

      if (existing.length > 0 && existing[0].status === 'ready') {
        if (existing[0].checksum === info.checksum) {
          await queryRunner.query('COMMIT');
          return { outcome: 'already_ready' };
        }
        throw new ConflictException({
          code: 'PART_CHECKSUM_MISMATCH',
          message: 'Part already uploaded with different checksum',
          expected_checksum: existing[0].checksum,
          received_checksum: info.checksum,
        });
      }

      if (existing.length > 0 && existing[0].status === 'receiving') {
        // 同一 owner 重试（客户端重发）：接管同一行继续
        if (existing[0].owner_token === info.ownerToken) {
          await queryRunner.query('COMMIT');
          return { outcome: 'claimed' };
        }
        // 他人在接收中：409，客户端稍后重试
        throw new ConflictException({
          code: 'PART_BEING_RECEIVED',
          message: 'Part is being received by another request; retry later',
        });
      }

      // 3) 行不存在：插入 receiving 行，获得接收权
      await queryRunner.query(
        `INSERT INTO upload_parts
           (upload_id, part_number, offset, size, checksum, status, owner_token, temp_name, received_at)
         VALUES (?, ?, ?, ?, ?, 'receiving', ?, ?, ?)`,
        [uploadId, partNumber, info.offset, info.size, info.checksum, info.ownerToken, info.tempName, info.now],
      );

      // 4) 会话状态 initiated -> uploading（幂等）
      await queryRunner.query(
        `UPDATE upload_sessions SET status = 'uploading' WHERE id = ? AND status = 'initiated'`,
        [uploadId],
      );

      await queryRunner.query('COMMIT');
      return { outcome: 'claimed' };
    } catch (error) {
      await queryRunner.query('ROLLBACK').catch(() => {});
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  /**
   * 阶段三：确认 receiving -> ready。带 owner_token + status 守卫，
   * 只有首次 receiving->ready 的事务才递增 received_size。
   */
  private async confirmPartReady(
    uploadId: string,
    partNumber: number,
    ownerToken: string,
    partSize: number,
  ): Promise<void> {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    try {
      await queryRunner.query('BEGIN IMMEDIATE');

      const result: any = await queryRunner.query(
        `UPDATE upload_parts
           SET status = 'ready', owner_token = NULL, temp_name = NULL, received_at = ?
         WHERE upload_id = ? AND part_number = ? AND owner_token = ? AND status = 'receiving'`,
        [Date.now(), uploadId, partNumber, ownerToken],
      );

      // better-sqlite3 / sqlite3 驱动返回 changes；不同驱动字段名兼容
      const affected = result?.changes ?? result?.affected ?? result?.rowCount ?? 0;

      if (affected === 1) {
        // 仅首次 receiving->ready 递增 received_size（重试/幂等路径不会走到这里，
        // 因为重试会在阶段一命中 already_ready 或 owner 不匹配）
        await queryRunner.query(
          `UPDATE upload_sessions SET received_size = received_size + ? WHERE id = ?`,
          [partSize, uploadId],
        );
        await queryRunner.query('COMMIT');
        return;
      }

      // affected === 0：行被超时回收任务回收了（owner 阶段二耗时超过阈值）
      await queryRunner.query('ROLLBACK');
      throw new ConflictException({
        code: 'PART_CLAIM_LOST',
        message: 'Part claim expired while writing; re-upload this part',
      });
    } catch (error) {
      await queryRunner.query('ROLLBACK').catch(() => {});
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  /**
   * checksum 不匹配时释放接收权：删除 receiving 行（带 owner 守卫，原生 SQL 单语句原子）。
   * 正式分块路径此时不存在（rename 未执行），无文件清理需求。
   */
  private async releasePartClaim(uploadId: string, partNumber: number, ownerToken: string): Promise<void> {
    await this.dataSource.query(
      `DELETE FROM upload_parts
       WHERE upload_id = ? AND part_number = ? AND owner_token = ? AND status = 'receiving'`,
      [uploadId, partNumber, ownerToken],
    );
  }
```

**并发语义证明（写进计划正文）**：

- **同 checksum 并发**：请求 A 在阶段一 INSERT 成功（claimed）；请求 B 的 INSERT 因 PK 冲突前先在 SELECT 分支看到 `receiving` 且 owner 不同 → 409 `PART_BEING_RECEIVED`；若 B 到达时 A 已 ready → `already_ready` 幂等返回。SQLite 单写者 + `BEGIN IMMEDIATE` 保证阶段一的 SELECT→INSERT 判定串行化。
- **不同 checksum 并发**：先到者 ready 后，后到者在阶段一看到 `ready` + checksum 不同 → 409 `PART_CHECKSUM_MISMATCH`，且**不删除任何文件**（v1.5 在事务内 deletePart 正式分块的 bug 消除）。
- **rename 覆盖**：只有 claimed owner 会执行 rename，且临时文件名含 owner uuid，两个 owner 的临时文件互不相同；输家永远不会 rename（阶段一已拒绝）。

## 4. completeUpload() 重写（替换 v1.5 第 2783–2972 行）

设计要点：

- 阶段一 `BEGIN IMMEDIATE`：`UPDATE ... SET status='verifying', verify_started_at=?, final_stored_name=COALESCE(final_stored_name, ?), final_file_id=COALESCE(final_file_id, ?) WHERE id=? AND status IN ('initiated','uploading')`。`final_stored_name` 与 `final_file_id` 在**抢占时**预生成并持久化——崩溃恢复与重试都使用同一路径/同一 files.id。
- 阶段二事务外：校验分块连续性（含 status='ready' 过滤）→ 用持久化的 `final_stored_name` 合并 → 校验最终 hash。
- 阶段三 `BEGIN IMMEDIATE`：`INSERT INTO files`（主键即预生成 `final_file_id`，PK 冲突即幂等命中）+ `UPDATE upload_sessions ... WHERE id=? AND status='verifying'`。两条语句同事务，天然幂等。

`apps/server/src/files/uploads.service.ts`（替换 completeUpload）：

```typescript
  async completeUpload(uploadId: string, uploadToken: string, finalHash?: string): Promise<{ file_id: string }> {
    // Token 校验（不限制状态，分支处理）
    const session = await this.validateUploadToken(uploadId, uploadToken);

    // 已完成：幂等返回
    if (session.status === UploadStatus.COMPLETED) {
      return { file_id: session.finalFileId! };
    }

    // ---------- 阶段一：BEGIN IMMEDIATE 抢占，预生成并持久化 final_stored_name / final_file_id ----------
    const preStoredName = this.storageService.generateStoredName();
    const preFileId = uuidv4();
    const verifyStartedAt = Date.now();

    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    let claimedSession: UploadSession;
    try {
      await queryRunner.query('BEGIN IMMEDIATE');

      const result: any = await queryRunner.query(
        `UPDATE upload_sessions
           SET status = 'verifying',
               verify_started_at = ?,
               final_stored_name = COALESCE(final_stored_name, ?),
               final_file_id = COALESCE(final_file_id, ?)
         WHERE id = ? AND status IN ('initiated', 'uploading')`,
        [verifyStartedAt, preStoredName, preFileId, uploadId],
      );
      const affected = result?.changes ?? result?.affected ?? 0;

      if (affected === 0) {
        const current = await queryRunner.query(
          `SELECT status, final_file_id FROM upload_sessions WHERE id = ?`,
          [uploadId],
        );
        if (current.length === 0) {
          throw new NotFoundException('Upload session not found');
        }
        if (current[0].status === 'completed') {
          await queryRunner.query('COMMIT');
          return { file_id: current[0].final_file_id };
        }
        if (current[0].status === 'verifying') {
          throw new ConflictException({
            code: 'UPLOAD_FINALIZING',
            message: 'Another request is completing this upload',
          });
        }
        throw new BadRequestException({
          code: 'INVALID_UPLOAD_STATE',
          message: `Cannot complete upload in state: ${current[0].status}`,
        });
      }

      // 抢占成功：读回持久化的最终名/文件 id（COALESCE 保证重试同路径）
      const rows = await queryRunner.query(
        `SELECT * FROM upload_sessions WHERE id = ?`,
        [uploadId],
      );
      await queryRunner.query('COMMIT');
      claimedSession = this.uploadsRepository.create(rows[0]) as UploadSession;
    } catch (error) {
      await queryRunner.query('ROLLBACK').catch(() => {});
      throw error;
    } finally {
      await queryRunner.release();
    }

    // ---------- 阶段二：事务外合并与校验（用持久化的 final_stored_name）----------
    const finalStoredName = claimedSession.finalStoredName!;
    const finalPath = this.storageService.getFinalPath(finalStoredName);
    try {
      // 仅统计 ready 分块；receiving 中的分块视为缺失（其 owner 在阶段一已无法就绪，
      // 因为 complete 抢占后 uploadPart 阶段一的会话状态守卫会拒绝新的 claim）
      const parts = await this.partsRepository.find({
        where: { uploadId, status: UploadPartStatus.READY },
        order: { partNumber: 'ASC' },
      });

      const totalParts = Math.ceil(claimedSession.expectedSize / claimedSession.chunkSize);
      if (parts.length !== totalParts) {
        throw new BadRequestException({
          code: 'MISSING_PARTS',
          message: `Missing parts: expected ${totalParts}, got ${parts.length}`,
        });
      }
      for (let i = 0; i < parts.length; i++) {
        if (parts[i].partNumber !== i || parts[i].offset !== i * claimedSession.chunkSize) {
          throw new BadRequestException({
            code: 'INVALID_PART_OFFSET',
            message: `Part ${i} has invalid offset`,
          });
        }
      }

      // 幂等安全：重试/恢复时 finalPath 相同，combineParts 覆盖写同一路径
      await this.storageService.combineParts(uploadId, totalParts, finalPath);

      const expectedHash = finalHash ?? claimedSession.expectedHash;
      if (expectedHash) {
        const actualHash = await this.storageService.calculateFileHash(finalPath);
        if (actualHash !== expectedHash) {
          throw new BadRequestException({
            code: 'FINAL_HASH_MISMATCH',
            message: 'Final file hash mismatch',
          });
        }
      }

      // ---------- 阶段三：BEGIN IMMEDIATE，INSERT files（幂等）+ 条件 UPDATE 会话 ----------
      const queryRunner3 = this.dataSource.createQueryRunner();
      await queryRunner3.connect();
      try {
        await queryRunner3.query('BEGIN IMMEDIATE');

        // files 主键即阶段一预生成的 final_file_id：
        // 重试/恢复时 PK 冲突 -> 捕获后视为已存在（幂等），不重复插入
        const fileRows = await queryRunner3.query(
          `SELECT id FROM files WHERE id = ?`,
          [claimedSession.finalFileId],
        );
        if (fileRows.length === 0) {
          const now = Date.now();
          await queryRunner3.query(
            `INSERT INTO files
               (id, folder_id, filename, stored_name, size, mime_type, hash_sha256,
                status, expires_at, uploaded_by_type, uploaded_by_id, download_count,
                created_at, updated_at)
             VALUES (?, NULL, ?, ?, ?, NULL, ?, 'active', ?, ?, ?, 0, ?, ?)`,
            [
              claimedSession.finalFileId,
              claimedSession.filename,
              finalStoredName,
              claimedSession.expectedSize,
              expectedHash ?? null,
              now + 24 * 60 * 60 * 1000, // 默认 24h（v1.6 另一修正项接入 default_expire_hours 设置）
              claimedSession.principalType,
              claimedSession.principalId,
              now,
              now,
            ],
          );
        }

        // 条件 UPDATE：只有仍处于 verifying 的会话才能被本请求完成；
        // affected=0 说明已被并发请求/恢复任务完成 -> 幂等返回即可
        await queryRunner3.query(
          `UPDATE upload_sessions
             SET status = 'completed', completed_at = ?, failure_reason = NULL
           WHERE id = ? AND status = 'verifying'`,
          [Date.now(), uploadId],
        );

        await queryRunner3.query('COMMIT');
      } catch (error) {
        await queryRunner3.query('ROLLBACK').catch(() => {});
        throw error;
      } finally {
        await queryRunner3.release();
      }

      // 清理临时目录：独立操作，失败不影响完成状态（孤儿由启动扫描兜底）
      this.storageService.deleteUploadTempDir(uploadId).catch(() => {});

      return { file_id: claimedSession.finalFileId! };
    } catch (error: unknown) {
      // 可恢复错误（分块缺失/hash 不匹配）：回退 uploading，保留分块等待重传；
      // final_stored_name/final_file_id 保持已持久化值，下次 complete 用同一组值
      let errorCode: string | undefined;
      if (error instanceof BadRequestException) {
        const response = error.getResponse();
        if (typeof response === 'object' && response !== null && 'code' in response) {
          errorCode = String((response as any).code);
        }
      }
      const recoverable = ['MISSING_PARTS', 'FINAL_HASH_MISMATCH', 'INVALID_PART_OFFSET'];
      if (errorCode && recoverable.includes(errorCode)) {
        await this.storageService.deleteFile(finalStoredName).catch(() => {}); // 删除半成品（事务外）
        await this.dataSource.query(
          `UPDATE upload_sessions SET status = 'uploading', verify_started_at = NULL
           WHERE id = ? AND status = 'verifying'`,
          [uploadId],
        );
      } else {
        // 不可恢复错误：标记 failed，保留 session 供排查；
        // 半成品文件由 FileLifecycleService 清理（status='failed' 且 final_stored_name 非空）
        await this.dataSource.query(
          `UPDATE upload_sessions SET status = 'failed', failure_reason = ?
           WHERE id = ? AND status = 'verifying'`,
          [error instanceof Error ? error.message : 'Unknown error', uploadId],
        );
      }
      throw error;
    }
  }
```

**并发 complete 语义**：两个并发请求都通过 `validateUploadToken` 后，阶段一的条件 UPDATE 只有先到者 `affected=1`；后到者 `affected=0` 且读到 `verifying` → 409 `UPLOAD_FINALIZING`；若读到 `completed`（先者已提交阶段三）→ 幂等返回 `file_id`。阶段三的 files INSERT 以 `final_file_id` 为主键，恢复任务与重试请求并发时先 SELECT 后 INSERT 在同一 `BEGIN IMMEDIATE` 事务内，不会重复插入。

## 5. FileLifecycleService：verifying 超时恢复 + 孤儿临时文件扫描（新建文件）

v1.5 只在 File Structure 列出 `file-lifecycle.service.ts` 而无实现。本节给出完整实现，包含 `recoverVerifyingUploads()`（@Cron 每分钟）与 `scanOrphanTempFiles()`（`onApplicationBootstrap`）。文件过期清理 `expireFiles()` 与删除清理 `purgeDeletingFiles()` 属于问题 4 范畴，此处仅保留接口位置注释。

`apps/server/src/files/file-lifecycle.service.ts`:

```typescript
import { Injectable, Logger, OnApplicationBootstrap } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { DataSource } from 'typeorm';
import { promises as fs } from 'fs';
import { join } from 'path';
import { StorageService } from './storage.service';

const VERIFYING_TIMEOUT_MS = 5 * 60 * 1000;   // verifying 超时 5 分钟
const RECEIVING_TIMEOUT_MS = 10 * 60 * 1000;  // receiving 分块超时 10 分钟（大于最长 8MB 写入预期）

@Injectable()
export class FileLifecycleService implements OnApplicationBootstrap {
  private readonly logger = new Logger(FileLifecycleService.name);

  constructor(
    private dataSource: DataSource,
    private storageService: StorageService,
  ) {}

  /**
   * 启动时扫描孤儿临时文件：
   * 1) temp/<upload_id>/ 目录但 upload_sessions 不存在 -> 整目录删除
   * 2) 会话为 completed/aborted/expired/failed -> 整目录删除
   * 3) 会话为 initiated/uploading/verifying -> 只删 *.part.tmp（owner 临时文件），保留 part_*.part
   */
  async onApplicationBootstrap(): Promise<void> {
    await this.scanOrphanTempFiles().catch((err) =>
      this.logger.error(`scanOrphanTempFiles failed: ${err}`),
    );
  }

  async scanOrphanTempFiles(): Promise<void> {
    const tempRoot = this.storageService.getTempRoot();
    let entries: string[];
    try {
      entries = await fs.readdir(tempRoot);
    } catch {
      return; // temp 根目录不存在
    }

    for (const uploadId of entries) {
      const dir = join(tempRoot, uploadId);
      const stat = await fs.stat(dir).catch(() => null);
      if (!stat?.isDirectory()) continue;

      const rows = await this.dataSource.query(
        `SELECT status FROM upload_sessions WHERE id = ?`,
        [uploadId],
      );

      if (rows.length === 0 || ['completed', 'aborted', 'expired', 'failed'].includes(rows[0].status)) {
        // 无会话或会话已终态：整目录删除
        await this.storageService.deleteUploadTempDir(uploadId).catch(() => {});
        continue;
      }

      // 活跃会话：仅删除 owner 临时文件（*.part.tmp），保留正式分块
      const files = await fs.readdir(dir).catch(() => [] as string[]);
      for (const name of files) {
        if (name.endsWith('.part.tmp')) {
          await fs.unlink(join(dir, name)).catch(() => {});
        }
      }
    }
  }

  /**
   * 每分钟：恢复卡在 verifying 超过 5 分钟的上传会话。
   * 判定：最终文件已存在（阶段二完成、阶段三前崩溃）-> 继续执行阶段三；
   *      最终文件不存在（阶段二中崩溃）-> 回退 uploading，客户端可重试 complete。
   */
  @Cron(CronExpression.EVERY_MINUTE)
  async recoverVerifyingUploads(): Promise<void> {
    const deadline = Date.now() - VERIFYING_TIMEOUT_MS;
    const stale = await this.dataSource.query(
      `SELECT id, final_stored_name, final_file_id FROM upload_sessions
       WHERE status = 'verifying' AND verify_started_at < ?`,
      [deadline],
    );

    for (const row of stale) {
      const { id: uploadId, final_stored_name: storedName, final_file_id: fileId } = row;
      if (!storedName || !fileId) {
        // 异常数据（不应发生：阶段一总是同时写三列），防御性回退
        await this.dataSource.query(
          `UPDATE upload_sessions SET status = 'uploading', verify_started_at = NULL
           WHERE id = ? AND status = 'verifying'`,
          [uploadId],
        );
        continue;
      }

      const exists = await this.storageService.fileExists(storedName);
      if (exists) {
        // 阶段二已完成：继续阶段三（与 completeUpload 阶段三完全同构，天然幂等）
        await this.finishVerifyingUpload(uploadId, storedName, fileId);
      } else {
        // 阶段二未完成：回退 uploading，等客户端重试 completeUpload（同一路径/同一 fileId）
        await this.dataSource.query(
          `UPDATE upload_sessions SET status = 'uploading', verify_started_at = NULL
           WHERE id = ? AND status = 'verifying'`,
          [uploadId],
        );
        this.logger.warn(`Upload ${uploadId} reverted verifying -> uploading (final file missing)`);
      }
    }
  }

  /**
   * 恢复任务执行的阶段三：INSERT files（幂等）+ 条件 UPDATE 会话。
   * 与客户端重试 completeUpload 并发时：双方都在 BEGIN IMMEDIATE 内先 SELECT files，
   * PK 为预生成 fileId，串行化后只有一个 INSERT 生效。
   */
  private async finishVerifyingUpload(uploadId: string, storedName: string, fileId: string): Promise<void> {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    try {
      await queryRunner.query('BEGIN IMMEDIATE');

      const fileRows = await queryRunner.query(`SELECT id FROM files WHERE id = ?`, [fileId]);
      if (fileRows.length === 0) {
        const sessionRows = await queryRunner.query(
          `SELECT filename, expected_size, expected_hash, principal_type, principal_id
           FROM upload_sessions WHERE id = ?`,
          [uploadId],
        );
        if (sessionRows.length === 0) {
          await queryRunner.query('ROLLBACK');
          return;
        }
        const s = sessionRows[0];
        const now = Date.now();
        await queryRunner.query(
          `INSERT INTO files
             (id, folder_id, filename, stored_name, size, mime_type, hash_sha256,
              status, expires_at, uploaded_by_type, uploaded_by_id, download_count,
              created_at, updated_at)
           VALUES (?, NULL, ?, ?, ?, NULL, ?, 'active', ?, ?, ?, 0, ?, ?)`,
          [
            fileId, s.filename, storedName, s.expected_size,
            s.expected_hash ?? null,
            now + 24 * 60 * 60 * 1000,
            s.principal_type, s.principal_id, now, now,
          ],
        );
      }

      // 条件 UPDATE：只认领仍处于 verifying 的会话
      await queryRunner.query(
        `UPDATE upload_sessions
           SET status = 'completed', completed_at = ?, failure_reason = NULL
         WHERE id = ? AND status = 'verifying'`,
        [Date.now(), uploadId],
      );

      await queryRunner.query('COMMIT');
      this.logger.log(`Recovered verifying upload ${uploadId} -> completed (file ${fileId})`);
    } catch (error) {
      await queryRunner.query('ROLLBACK').catch(() => {});
      throw error;
    } finally {
      await queryRunner.release();
    }

    // 事务外清理临时目录
    this.storageService.deleteUploadTempDir(uploadId).catch(() => {});
  }

  /**
   * 每分钟：回收超时 receiving 分块（owner 阶段二崩溃留下的行）。
   * 删除 receiving 行 + 对应 owner 临时文件；会话状态不变，客户端可重传该分块。
   * 必须与 confirmPartReady 的 owner 判定一致：只删 received_at 超时的 receiving 行。
   */
  @Cron(CronExpression.EVERY_MINUTE)
  async recoverStaleReceivingParts(): Promise<void> {
    const deadline = Date.now() - RECEIVING_TIMEOUT_MS;
    const stale = await this.dataSource.query(
      `SELECT upload_id, part_number, owner_token FROM upload_parts
       WHERE status = 'receiving' AND received_at < ?`,
      [deadline],
    );

    for (const part of stale) {
      // 单语句原子删除（条件重复守卫，防止与 confirmPartReady 竞态：
      // 若 owner 刚好完成 ready 转换，此 DELETE 不匹配任何行）
      const result: any = await this.dataSource.query(
        `DELETE FROM upload_parts
         WHERE upload_id = ? AND part_number = ? AND status = 'receiving'
           AND owner_token = ? AND received_at < ?`,
        [part.upload_id, part.part_number, part.owner_token, deadline],
      );
      const affected = result?.changes ?? result?.affected ?? 0;
      if (affected === 1 && part.owner_token) {
        // 事务外删除该 owner 的临时文件（可能不存在，幂等）
        await this.storageService
          .deleteTempPart(part.upload_id, part.part_number, part.owner_token)
          .catch(() => {});
      }
    }
  }

  // expireFiles() / purgeDeletingFiles()：见问题 4（文件生命周期）修正项，此处不重复定义。
}
```

/**
 * Fake QueryRunner：在内存 Map 上模拟 upload_parts/upload_sessions 的 SQL 语义。
 * 关键：BEGIN IMMEDIATE 期间给 fake 加互斥锁（single-writer），
 * 让并发单测真实暴露"非原子判定"问题，而不是靠 mock 返回编排。
 */
function createFakeDb() {
  const parts = new Map<string, any>();        // key = `${upload_id}:${part_number}`
  const sessions = new Map<string, any>();     // key = id
  const files = new Map<string, any>();
  let writeLock: Promise<void> = Promise.resolve();

  const acquire = () => {
    const prev = writeLock;
    let release!: () => void;
    writeLock = new Promise<void>((r) => (release = r));
    return prev.then(() => release);
  };

  const query = jest.fn(async (sql: string, params: any[] = []) => {
    if (sql === 'BEGIN IMMEDIATE') { (query as any).__release = await acquire(); return; }
    if (sql === 'COMMIT' || sql === 'ROLLBACK') { (query as any).__release?.(); return; }

    // upload_parts INSERT（模拟 PK 冲突）
    if (sql.startsWith('INSERT INTO upload_parts')) {
      const key = `${params[0]}:${params[1]}`;
      if (parts.has(key)) { const e: any = new Error('UNIQUE constraint failed'); e.code = 'SQLITE_CONSTRAINT_PRIMARYKEY'; throw e; }
      parts.set(key, { upload_id: params[0], part_number: params[1], offset: params[2], size: params[3],
        checksum: params[4], status: 'receiving', owner_token: params[5], temp_name: params[6], received_at: params[7] });
      return { changes: 1 };
    }
    // upload_parts SELECT by pk
    if (sql.startsWith('SELECT status, checksum, owner_token FROM upload_parts')) {
      const row = parts.get(`${params[0]}:${params[1]}`);
      return row ? [{ status: row.status, checksum: row.checksum, owner_token: row.owner_token }] : [];
    }
    // upload_parts UPDATE receiving->ready（带 owner 守卫）
    if (sql.startsWith('UPDATE upload_parts')) {
      const [, uploadId, partNumber, ownerToken] = params;
      const row = parts.get(`${uploadId}:${partNumber}`);
      if (row && row.owner_token === ownerToken && row.status === 'receiving') {
        row.status = 'ready'; row.owner_token = null; row.temp_name = null; row.received_at = params[0];
        return { changes: 1 };
      }
      return { changes: 0 };
    }
    // upload_parts DELETE（releasePartClaim / recoverStaleReceivingParts）
    if (sql.startsWith('DELETE FROM upload_parts')) {
      const row = parts.get(`${params[0]}:${params[1]}`);
      if (row && row.status === 'receiving' && (params.length < 3 || row.owner_token === params[2])) {
        parts.delete(`${params[0]}:${params[1]}`);
        return { changes: 1 };
      }
      return { changes: 0 };
    }
    // upload_sessions SELECT status / SELECT * / UPDATE 各形态按 v1.6 SQL 文本逐一模拟
    // （条件 UPDATE：解析 WHERE 中的 status 守卫，不满足返回 { changes: 0 }）
    // files SELECT/INSERT 同理（PK 冲突抛 SQLITE_CONSTRAINT_PRIMARYKEY）
    // ...（计划中附完整实现，此处列骨架）
    return [];
  });

  const dataSource: any = {
    createQueryRunner: () => ({
      connect: jest.fn(),
      release: jest.fn(),
      query,
    }),
    query, // releasePartClaim / lifecycle 直接用 dataSource.query
  };
  return { dataSource, parts, sessions, files };
}
