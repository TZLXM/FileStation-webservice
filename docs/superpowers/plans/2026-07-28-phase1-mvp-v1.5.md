# FileStation Phase 1 MVP Implementation Plan v1.5

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立 FileStation MVP：项目脚手架、安全初始化、账号密码登录+JWT、文件上传下载、文件夹管理、文件有效期、分享链接、基础设置页。

**Architecture:** NestJS 后端 + React 前端 + SQLite 数据库，npm workspaces monorepo。后端模块化设计，前端组件化开发，TDD 驱动。

**Tech Stack:** NestJS 10, React 18, TypeORM 0.3, SQLite 3, Vite 5, shadcn/ui, Tailwind CSS 3, Jest, React Testing Library

## Global Constraints

- Node.js >= 20.0.0
- SQLite WAL 模式，busy_timeout=5000
- 禁用 `SELECT ... FOR UPDATE`（SQLite 不支持）
- 写事务中禁止文件 I/O 或网络请求
- 所有密码使用 bcrypt 哈希
- 时间字段：Unix 毫秒 (INTEGER)，UTC
- API 版本：`/api/v1/*`（Controller 不含前缀，仅 main.ts 设置 globalPrefix）
- 上传分块大小：8MB (8388608 bytes)
- 初始化 Token 有效期：10 分钟，自动生成，单次使用，恒定时间比较
- JWT 有效期：24 小时，Refresh Token：7 天，HttpOnly Cookie 存储
- 禁止 `Access-Control-Allow-Origin: *`
- **Phase 1 禁止 direct 分享类型**（Phase 3 实现）
- **v1.5 修正**：初始化使用原生 SQL 事务（BEGIN IMMEDIATE + 纯 SQL 操作）
- **v1.5 修正**：ensureInitToken() 每次启动覆盖旧 Token
- **v1.5 修正**：恒定时间比较使用 timingSafeEqual
- **v1.5 修正**：分块并发安全（upload_parts 增加 status/owner_token/temp_name）
- **v1.5 修正**：下载票据完整实现（download_tickets 表 + /downloads/:ticket 端点）
- **v1.5 修正**：文件生命周期管理（FileLifecycleService + @Cron）

```typescript
import { Module, OnApplicationBootstrap, Injectable, Logger } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { DataSource } from 'typeorm';

@Injectable()
export class DatabaseInitializer implements OnApplicationBootstrap {
  private readonly logger = new Logger(DatabaseInitializer.name);

  constructor(private readonly dataSource: DataSource) {}

  async onApplicationBootstrap(): Promise<void> {
    // 执行 PRAGMA 配置
    await this.dataSource.query('PRAGMA journal_mode = WAL');
    await this.dataSource.query('PRAGMA busy_timeout = 5000');
    await this.dataSource.query('PRAGMA foreign_keys = ON');

    // 验证配置
    const [journalMode] = await this.dataSource.query('PRAGMA journal_mode');
    const [busyTimeout] = await this.dataSource.query('PRAGMA busy_timeout');
    const [foreignKeys] = await this.dataSource.query('PRAGMA foreign_keys');

    this.logger.log(`SQLite journal_mode: ${journalMode.journal_mode}`);
    this.logger.log(`SQLite busy_timeout: ${busyTimeout.busy_timeout}`);
    this.logger.log(`SQLite foreign_keys: ${foreignKeys.foreign_keys}`);

    if (foreignKeys.foreign_keys !== 1) {
      throw new Error('SQLite foreign keys are disabled');
    }
  }
}
```

`apps/server/src/auth/auth.service.ts`:
```typescript
import { Injectable, UnauthorizedException, BadRequestException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { ConfigService } from '@nestjs/config';
import { AccountsService } from '../accounts/accounts.service';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, DataSource } from 'typeorm';
import { Session } from './entities/session.entity';
import { LoginChallenge } from './entities/login-challenge.entity';
import { SystemMeta } from './entities/system-meta.entity';
import { v4 as uuidv4 } from 'uuid';
import { createHash, timingSafeEqual } from 'crypto';
import { LoginRequest, TokenPair, JwtPayload } from '@filestation/shared';

@Injectable()
export class AuthService {
  constructor(
    private accountsService: AccountsService,
    private jwtService: JwtService,
    private configService: ConfigService,
    @InjectRepository(Session)
    private sessionsRepository: Repository<Session>,
    @InjectRepository(LoginChallenge)
    private challengesRepository: Repository<LoginChallenge>,
    @InjectRepository(SystemMeta)
    private systemMetaRepository: Repository<SystemMeta>,
    private dataSource: DataSource,
  ) {}

  async isInitialized(): Promise<boolean> {
    const meta = await this.systemMetaRepository.findOne({ where: { key: 'initialized_at' } });
    return !!meta;
  }

  async generateInitToken(): Promise<string> {
    // 生成 32 字符随机 Token
    const token = uuidv4().replace(/-/g, '') + uuidv4().replace(/-/g, '').substring(0, 8);
    const expiresAt = Date.now() + 10 * 60 * 1000; // 10 minutes
    const tokenHash = createHash('sha256').update(token).digest('hex');

    await this.systemMetaRepository.save({
      key: 'init_token',
      value: JSON.stringify({ token_hash: tokenHash, expires_at: expiresAt }),
    });

    // 完整 Token 只返回一次，不写入数据库
    return token;
  }

  async ensureInitToken(): Promise<string | null> {
    // 检查是否已初始化
    if (await this.isInitialized()) {
      return null;
    }

    // 每次启动都覆盖旧 Token（防重启后无法获取）
    return this.generateInitToken();
  }

  async initialize(username: string, password: string, initToken: string): Promise<TokenPair> {
    // 密码哈希在事务外完成（避免长时间持有写锁）
    const passwordHash = await import('bcrypt').then(b => b.hash(password, 10));

    // 使用原生 BEGIN IMMEDIATE 事务（纯 SQL 操作）
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();

    try {
      await queryRunner.query('BEGIN IMMEDIATE');

      // 事务内重新验证初始化 Token（原生 SQL）
      const metaResult = await queryRunner.query(
        `SELECT value FROM system_meta WHERE key = 'init_token'`,
      );
      if (metaResult.length === 0) {
        throw new UnauthorizedException('No initialization token generated');
      }

      const { token_hash: expectedTokenHash, expires_at: expiresAt } = JSON.parse(metaResult[0].value);
      
      // 检查过期
      if (Date.now() > expiresAt) {
        throw new UnauthorizedException('Initialization token expired');
      }

      // 恒定时间比较（使用 timingSafeEqual）
      const initTokenHash = createHash('sha256').update(initToken).digest('hex');
      const actualBuffer = Buffer.from(initTokenHash, 'hex');
      const expectedBuffer = Buffer.from(expectedTokenHash, 'hex');
      
      if (actualBuffer.length !== expectedBuffer.length || !timingSafeEqual(actualBuffer, expectedBuffer)) {
        throw new UnauthorizedException('Invalid initialization token');
      }

      // 事务内重新检查是否已初始化（原生 SQL）
      const initializedResult = await queryRunner.query(
        `SELECT 1 FROM system_meta WHERE key = 'initialized_at'`,
      );
      if (initializedResult.length > 0) {
        throw new BadRequestException('System already initialized');
      }

      // 创建账户（原生 SQL）
      const now = Date.now();
      const accountId = uuidv4();
      await queryRunner.query(
        `INSERT INTO admin_accounts (id, username, password_hash, password_changed_at, created_at, is_active) VALUES (?, ?, ?, ?, ?, ?)`,
        [accountId, username, passwordHash, now, now, 1],
      );

      // 标记已初始化（原生 SQL）
      await queryRunner.query(
        `INSERT INTO system_meta (key, value) VALUES ('initialized_at', ?)`,
        [now.toString()],
      );
      await queryRunner.query(
        `INSERT INTO system_meta (key, value) VALUES ('first_account_id', ?)`,
        [accountId],
      );

      // 删除初始化 Token（单次使用）
      await queryRunner.query(
        `DELETE FROM system_meta WHERE key = 'init_token'`,
      );

      await queryRunner.query('COMMIT');

      // 生成 Token
      return this.generateTokens(accountId, username);
    } catch (error) {
      await queryRunner.query('ROLLBACK').catch(() => {});
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  async login(loginDto: LoginRequest): Promise<TokenPair> {
    const account = await this.accountsService.findByUsername(loginDto.username);
    if (!account) {
      throw new UnauthorizedException('Invalid credentials');
    }

    const isValid = await this.accountsService.validatePassword(account, loginDto.password);
    if (!isValid) {
      throw new UnauthorizedException('Invalid credentials');
    }

    // Phase 1 简化：直接返回 Token，不检查 TOTP
    return this.generateTokens(account.id, account.username);
  }

  async refreshToken(refreshToken: string): Promise<TokenPair> {
    const tokenHash = this.hashToken(refreshToken);
    
    // 原子抢占轮换权（明确 SQL，非 TypeORM null 查询）
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    await queryRunner.startTransaction();

    try {
      // 原子更新：仅当未吊销且未过期时
      const result = await queryRunner.manager
        .createQueryBuilder()
        .update(Session)
        .set({ revokedAt: Date.now() })
        .where('refresh_token_hash = :tokenHash', { tokenHash })
        .andWhere('revoked_at IS NULL')
        .andWhere('expires_at > :now', { now: Date.now() })
        .execute();

      if (result.affected === 0) {
        throw new UnauthorizedException('Invalid or expired refresh token');
      }

      // 查询会话获取账户
      const session = await queryRunner.manager.findOne(Session, {
        where: { refreshTokenHash: tokenHash },
      });

      if (!session) {
        throw new UnauthorizedException('Session not found');
      }

      const account = await this.accountsService.findById(session.accountId);
      if (!account) {
        throw new UnauthorizedException('Account not found');
      }

      // 生成新 Token（在同一事务中）
      const tokens = await this.generateTokensInTransaction(queryRunner, account.id, account.username);

      await queryRunner.commitTransaction();
      return tokens;
    } catch (error) {
      await queryRunner.rollbackTransaction();
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  private async generateTokensInTransaction(queryRunner: any, accountId: string, username: string): Promise<TokenPair> {
    const payload: Omit<JwtPayload, 'iat' | 'exp'> = {
      sub: accountId,
      username,
      principal_type: 'admin',
    };

    const accessToken = this.jwtService.sign(payload);
    const refreshToken = uuidv4();
    const expiresIn = 24 * 60 * 60;

    const session = queryRunner.manager.create(Session, {
      id: uuidv4(),
      accountId,
      refreshTokenHash: this.hashToken(refreshToken),
      createdAt: Date.now(),
      expiresAt: Date.now() + 7 * 24 * 60 * 60 * 1000,
    });
    await queryRunner.manager.save(session);

    return { accessToken, refreshToken, expiresIn };
  }

  async logout(refreshToken: string): Promise<void> {
    const tokenHash = this.hashToken(refreshToken);
    await this.sessionsRepository.update(
      { refreshTokenHash: tokenHash },
      { revokedAt: Date.now() },
    );
  }

  private async generateTokens(accountId: string, username: string): Promise<{ accessToken: string; refreshToken: string; expiresIn: number }> {
    const payload: Omit<JwtPayload, 'iat' | 'exp'> = {
      sub: accountId,
      username,
      principal_type: 'admin',
    };

    const accessToken = this.jwtService.sign(payload);
    const refreshToken = uuidv4();
    const expiresIn = 24 * 60 * 60; // 24 hours in seconds

    // 保存会话
    const session = this.sessionsRepository.create({
      id: uuidv4(),
      accountId,
      refreshTokenHash: this.hashToken(refreshToken),
      createdAt: Date.now(),
      expiresAt: Date.now() + 7 * 24 * 60 * 60 * 1000, // 7 days
    });
    await this.sessionsRepository.save(session);

    return { accessToken, refreshToken, expiresIn };
  }

  private hashToken(token: string): string {
    return createHash('sha256').update(token).digest('hex');
  }
}
```

```typescript
import { Response, Request } from 'express';

@Controller('auth')
export class AuthController {

  @Get('status')

      message: 'Success',
      data: { initialized },

    };

        access_token: result.accessToken,
        expires_in: result.expiresIn,
      },
      request_id: crypto.randomUUID(),
    };

  @HttpCode(HttpStatus.OK)

        access_token: result.accessToken,
        expires_in: result.expiresIn,
      },

      code: 'OK',
      message: 'Logged out successfully',
      data: null,
      request_id: crypto.randomUUID(),
```

```typescript
import { Injectable, BadRequestException, InternalServerErrorException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createWriteStream, createReadStream, promises as fs } from 'fs';
import { join } from 'path';
import { v4 as uuidv4 } from 'uuid';
import { createHash } from 'crypto';

@Injectable()
export class StorageService {
  private readonly storagePath: string;
  private readonly tempPath: string;

  constructor(private configService: ConfigService) {
    this.storagePath = this.configService.get<string>('app.storagePath') || './data/storage';
    this.tempPath = this.configService.get<string>('app.tempPath') || './data/temp';
  }

  async ensureDirectories(): Promise<void> {
    await fs.mkdir(this.storagePath, { recursive: true });
    await fs.mkdir(this.tempPath, { recursive: true });
  }

  getUploadTempDir(uploadId: string): string {
    return join(this.tempPath, uploadId);
  }

  getPartPath(uploadId: string, partNumber: number): string {
    return join(this.tempPath, uploadId, `part_${partNumber.toString().padStart(6, '0')}.part`);
  }

  getPartTempPath(uploadId: string, partNumber: number, uniqueId: string): string {
    return join(this.tempPath, uploadId, `part_${partNumber.toString().padStart(6, '0')}.${uniqueId}.part.tmp`);
  }

  getFinalPath(storedName: string): string {
    return join(this.storagePath, storedName);
  }

  async createUploadTempDir(uploadId: string): Promise<void> {
    const dir = this.getUploadTempDir(uploadId);
    await fs.mkdir(dir, { recursive: true });
  }

  async writePart(uploadId: string, partNumber: number, uniqueId: string, data: Buffer, expectedChecksum: string): Promise<void> {
    const tempPath = this.getPartTempPath(uploadId, partNumber, uniqueId);
    const finalPath = this.getPartPath(uploadId, partNumber);

    // 使用文件句柄确保 fsync
    const handle = await fs.open(tempPath, 'w');
    try {
      await handle.writeFile(data);
      await handle.sync(); // 真实 fsync
    } finally {
      await handle.close();
    }

    // 校验 checksum
    const actualChecksum = createHash('sha256').update(data).digest('hex');
    if (actualChecksum !== expectedChecksum) {
      await fs.unlink(tempPath).catch(() => {});
      throw new BadRequestException({
        code: 'CHECKSUM_MISMATCH',
        message: 'Part checksum mismatch',
        expected: expectedChecksum,
        actual: actualChecksum,
      });
    }

    // 原子重命名
    await fs.rename(tempPath, finalPath);
  }

  async readPart(uploadId: string, partNumber: number): Promise<Buffer> {
    const path = this.getPartPath(uploadId, partNumber);
    return fs.readFile(path);
  }

  async deletePart(uploadId: string, partNumber: number): Promise<void> {
    const path = this.getPartPath(uploadId, partNumber);
    await fs.unlink(path).catch(() => {});
  }

  async deleteUploadTempDir(uploadId: string): Promise<void> {
    const dir = this.getUploadTempDir(uploadId);
    await fs.rm(dir, { recursive: true, force: true });
  }

  async combineParts(uploadId: string, totalParts: number, finalPath: string): Promise<void> {
    const writeStream = createWriteStream(finalPath);

    try {
      for (let i = 0; i < totalParts; i++) {
        const partPath = this.getPartPath(uploadId, i);
        const readStream = createReadStream(partPath);
        
        await new Promise<void>((resolve, reject) => {
          readStream.pipe(writeStream, { end: false });
          readStream.on('end', resolve);
          readStream.on('error', reject);
        });
      }
    } finally {
      writeStream.end();
      await new Promise<void>((resolve) => writeStream.on('finish', resolve));
      
      // fsync 最终文件
      const handle = await fs.open(finalPath, 'r+');
      try {
        await handle.sync();
      } finally {
        await handle.close();
      }
    }
  }

  async moveToFinal(tempPath: string, storedName: string): Promise<void> {
    const finalPath = this.getFinalPath(storedName);
    await fs.rename(tempPath, finalPath);
  }

  async deleteFile(storedName: string): Promise<void> {
    const path = this.getFinalPath(storedName);
    await fs.unlink(path).catch(() => {});
  }

  async getFileStream(storedName: string, start?: number, end?: number) {
    const path = this.getFinalPath(storedName);
    const options: any = {};
    if (start !== undefined) options.start = start;
    if (end !== undefined) options.end = end;
    return createReadStream(path, options);
  }

  async getFileStats(storedName: string) {
    const path = this.getFinalPath(storedName);
    return fs.stat(path);
  }

  async fileExists(storedName: string): Promise<boolean> {
    try {
      await fs.access(this.getFinalPath(storedName));
      return true;
    } catch {
      return false;
    }
  }

  generateStoredName(): string {
    return uuidv4();
  }

  async calculateFileHash(filePath: string): Promise<string> {
    return new Promise((resolve, reject) => {
      const hash = createHash('sha256');
      const stream = createReadStream(filePath);
      stream.on('data', (data) => hash.update(data));
      stream.on('end', () => resolve(hash.digest('hex')));
      stream.on('error', reject);
    });
  }
}
```

`apps/server/src/files/uploads.service.ts`:
```typescript
import { Injectable, BadRequestException, NotFoundException, ConflictException, GoneException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, DataSource, In } from 'typeorm';
import { UploadSession, UploadStatus } from './entities/upload-session.entity';
import { UploadPart } from './entities/upload-part.entity';
import { File, FileStatus } from './entities/file.entity';
import { StorageService } from './storage.service';
import { v4 as uuidv4 } from 'uuid';
import { createHash } from 'crypto';
import { UploadInitRequest, UploadInitResponse, UploadStatus as UploadStatusType } from '@filestation/shared';

@Injectable()
export class UploadsService {
  constructor(
    @InjectRepository(UploadSession)
    private uploadsRepository: Repository<UploadSession>,
    @InjectRepository(UploadPart)
    private partsRepository: Repository<UploadPart>,
    @InjectRepository(File)
    private filesRepository: Repository<File>,
    private storageService: StorageService,
    private dataSource: DataSource,
  ) {}

  async initializeUpload(
    request: UploadInitRequest,
    principalType: 'admin' | 'temp_code',
    principalId: string,
  ): Promise<UploadInitResponse> {
    const uploadId = uuidv4();
    const uploadToken = uuidv4();
    const chunkSize = request.chunk_size || 8 * 1024 * 1024; // 8MB default
    const now = Date.now();

    // 创建临时目录
    await this.storageService.createUploadTempDir(uploadId);

    const session = this.uploadsRepository.create({
      id: uploadId,
      uploadTokenHash: this.hashToken(uploadToken),
      filename: request.filename,
      expectedSize: request.size,
      expectedHash: request.hash || null,
      chunkSize,
      status: UploadStatus.INITIATED,
      receivedSize: 0,
      tempPath: this.storageService.getUploadTempDir(uploadId),
      principalType,
      principalId,
      createdAt: now,
      expiresAt: now + 24 * 60 * 60 * 1000, // 24 hours
    });

    await this.uploadsRepository.save(session);

    return {
      upload_id: uploadId,
      upload_token: uploadToken,
      chunk_size: chunkSize,
      expires_at: new Date(session.expiresAt).toISOString(),
    };
  }

  async uploadPart(
    uploadId: string,
    partNumber: number,
    data: Buffer,
    checksum: string,
    uploadToken: string,
  ): Promise<{ received: boolean; part_checksum: string }> {
    // 验证 Token + 状态（仅 initiated/uploading 可上传分块）
    const session = await this.validateUploadToken(uploadId, uploadToken, [
      UploadStatus.INITIATED,
      UploadStatus.UPLOADING,
    ]);

    // 验证分块编号
    const totalParts = Math.ceil(session.expectedSize / session.chunkSize);
    if (partNumber < 0 || partNumber >= totalParts) {
      throw new BadRequestException({
        code: 'INVALID_PART_NUMBER',
        message: `Part number must be between 0 and ${totalParts - 1}`,
      });
    }

    // 验证分块大小
    const expectedSize = partNumber === totalParts - 1
      ? session.expectedSize - partNumber * session.chunkSize
      : session.chunkSize;
    if (data.length !== expectedSize) {
      throw new BadRequestException({
        code: 'INVALID_PART_SIZE',
        message: `Part size must be ${expectedSize} bytes`,
        expected: expectedSize,
        actual: data.length,
      });
    }

    // 检查是否已存在（幂等）
    const existingPart = await this.partsRepository.findOne({
      where: { uploadId, partNumber },
    });

    if (existingPart) {
      if (existingPart.checksum === checksum) {
        return { received: true, part_checksum: checksum };
      } else {
        throw new ConflictException({
          code: 'PART_CHECKSUM_MISMATCH',
          message: 'Part already exists with different checksum',
          expected_checksum: existingPart.checksum,
          received_checksum: checksum,
        });
      }
    }

    // 生成唯一临时文件 ID（防并发覆盖）
    const uniqueId = uuidv4();

    // 写入存储（带 fsync，独立临时文件）
    await this.storageService.writePart(uploadId, partNumber, uniqueId, data, checksum);

    // 记录分块（事务，原子更新 received_size）
    await this.dataSource.transaction(async (manager) => {
      // 再次检查是否已存在（事务内，防并发）
      const existing = await manager.findOne(UploadPart, {
        where: { uploadId, partNumber },
      });

      if (existing) {
        if (existing.checksum === checksum) {
          // 已存在相同分块，删除临时文件
          await this.storageService.deletePart(uploadId, partNumber).catch(() => {});
          return;
        } else {
          throw new ConflictException({
            code: 'PART_CHECKSUM_MISMATCH',
            message: 'Part already exists with different checksum',
            expected_checksum: existing.checksum,
            received_checksum: checksum,
          });
        }
      }

      const part = manager.create(UploadPart, {
        uploadId,
        partNumber,
        offset: partNumber * session.chunkSize,
        size: data.length,
        checksum,
        receivedAt: Date.now(),
      });
      await manager.save(part);

      // 原子递增 received_size
      await manager.increment(UploadSession, { id: uploadId }, 'receivedSize', data.length);
      await manager.update(UploadSession, uploadId, {
        status: UploadStatus.UPLOADING,
      });
    });

    return { received: true, part_checksum: checksum };
  }

  async getUploadStatus(uploadId: string, uploadToken: string): Promise<UploadStatusType> {
    const session = await this.validateUploadToken(uploadId, uploadToken);

    const parts = await this.partsRepository.find({
      where: { uploadId },
      order: { partNumber: 'ASC' },
    });

    const totalParts = Math.ceil(session.expectedSize / session.chunkSize);

    return {
      id: session.id,
      status: session.status as UploadStatusType['status'],
      received_parts: parts.map((p) => p.partNumber),
      received_size: session.receivedSize,
      total_parts: totalParts,
      expected_size: session.expectedSize,
    };
  }

  async completeUpload(uploadId: string, uploadToken: string, finalHash?: string): Promise<{ file_id: string }> {
    // 验证 Token（不限制状态，根据状态分支处理）
    const session = await this.validateUploadToken(uploadId, uploadToken);

    // 根据状态分支处理
    if (session.status === UploadStatus.COMPLETED) {
      return { file_id: session.finalFileId! };
    }

    if (session.status === UploadStatus.VERIFYING) {
      throw new ConflictException({
        code: 'UPLOAD_FINALIZING',
        message: 'Another request is completing this upload',
      });
    }

    if (session.status !== UploadStatus.INITIATED && session.status !== UploadStatus.UPLOADING) {
      throw new BadRequestException({
        code: 'INVALID_UPLOAD_STATE',
        message: `Cannot complete upload in state: ${session.status}`,
      });
    }

    // 阶段一：短事务抢占（使用 In 条件）
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    await queryRunner.startTransaction();

    let sessionToComplete: UploadSession;
    try {
      // 条件更新抢占完成权
      const result = await queryRunner.manager.update(
        UploadSession,
        {
          id: uploadId,
          status: In([UploadStatus.INITIATED, UploadStatus.UPLOADING]),
        },
        {
          status: UploadStatus.VERIFYING,
          verifyStartedAt: Date.now(),
        },
      );

      if (result.affected === 0) {
        // 检查当前状态（可能并发修改）
        const current = await queryRunner.manager.findOne(UploadSession, { where: { id: uploadId } });
        if (!current) {
          throw new NotFoundException('Upload session not found');
        }
        if (current.status === UploadStatus.COMPLETED) {
          return { file_id: current.finalFileId! };
        }
        if (current.status === UploadStatus.VERIFYING) {
          throw new ConflictException({
            code: 'UPLOAD_FINALIZING',
            message: 'Another request is completing this upload',
          });
        }
        throw new BadRequestException({
          code: 'INVALID_UPLOAD_STATE',
          message: `Cannot complete upload in state: ${current.status}`,
        });
      }

      sessionToComplete = await queryRunner.manager.findOne(UploadSession, { where: { id: uploadId } }) as UploadSession;
      await queryRunner.commitTransaction();
    } catch (error) {
      await queryRunner.rollbackTransaction();
      throw error;
    } finally {
      await queryRunner.release();
    }

    // 阶段二：事务外处理
    let finalStoredName: string | null = null;
    try {
      // 验证分块连续性
      const parts = await this.partsRepository.find({
        where: { uploadId },
        order: { partNumber: 'ASC' },
      });

      const totalParts = Math.ceil(sessionToComplete.expectedSize / sessionToComplete.chunkSize);
      if (parts.length !== totalParts) {
        throw new BadRequestException({
          code: 'MISSING_PARTS',
          message: `Missing parts: expected ${totalParts}, got ${parts.length}`,
        });
      }

      // 验证 offset 连续性
      for (let i = 0; i < parts.length; i++) {
        const expectedOffset = i * sessionToComplete.chunkSize;
        if (parts[i].offset !== expectedOffset) {
          throw new BadRequestException({
            code: 'INVALID_PART_OFFSET',
            message: `Part ${i} has invalid offset: expected ${expectedOffset}, got ${parts[i].offset}`,
          });
        }
      }

      // 生成最终存储名
      finalStoredName = this.storageService.generateStoredName();
      const finalPath = this.storageService.getFinalPath(finalStoredName);

      // 合并分块（等待流完成）
      await this.storageService.combineParts(uploadId, totalParts, finalPath);

      // 验证最终哈希（优先使用初始化时提供的 expected_hash）
      const expectedHash = finalHash ?? sessionToComplete.expectedHash;
      if (expectedHash) {
        const actualHash = await this.storageService.calculateFileHash(finalPath);
        if (actualHash !== expectedHash) {
          await this.storageService.deleteFile(finalStoredName);
          throw new BadRequestException({
            code: 'FINAL_HASH_MISMATCH',
            message: 'Final file hash mismatch',
          });
        }
      }

      // 阶段三：短事务完成（返回 file.id）
      const completedFileId = await this.dataSource.transaction(async (manager) => {
        // 创建文件记录
        const file = manager.create(File, {
          id: uuidv4(),
          folderId: null, // 默认收件箱
          filename: sessionToComplete.filename,
          storedName: finalStoredName,
          size: sessionToComplete.expectedSize,
          mimeType: null,
          hashSha256: expectedHash || null,
          status: FileStatus.ACTIVE,
          expiresAt: Date.now() + 24 * 60 * 60 * 1000, // 默认 24 小时过期
          uploadedByType: sessionToComplete.principalType,
          uploadedById: sessionToComplete.principalId,
          createdAt: Date.now(),
          updatedAt: Date.now(),
        });
        await manager.save(file);

        // 更新上传会话
        await manager.update(UploadSession, uploadId, {
          status: UploadStatus.COMPLETED,
          finalFileId: file.id,
          finalStoredName,
          completedAt: Date.now(),
        });

        return file.id;
      });

      // 清理临时分块（独立操作，失败不影响完成状态）
      this.storageService.deleteUploadTempDir(uploadId).catch((err) => {
        console.warn(`Failed to cleanup upload temp dir ${uploadId}:`, err);
      });

      return { file_id: completedFileId };
    } catch (error: unknown) {
      // 阶段三失败：删除孤立正式文件
      if (finalStoredName) {
        await this.storageService.deleteFile(finalStoredName).catch(() => {});
      }

      // 可恢复错误：恢复为 uploading 状态（读取 error.getResponse()）
      let errorCode: string | undefined;
      if (error instanceof BadRequestException) {
        const response = error.getResponse();
        if (typeof response === 'object' && response !== null && 'code' in response) {
          errorCode = String((response as any).code);
        }
      }

      const recoverableErrors = ['MISSING_PARTS', 'FINAL_HASH_MISMATCH', 'INVALID_PART_OFFSET'];
      const isRecoverable = errorCode && recoverableErrors.includes(errorCode);

      if (isRecoverable) {
        await this.uploadsRepository.update(uploadId, {
          status: UploadStatus.UPLOADING,
        });
      } else {
        // 不可恢复错误：标记失败
        await this.uploadsRepository.update(uploadId, {
          status: UploadStatus.FAILED,
          failureReason: error instanceof Error ? error.message : 'Unknown error',
        });
      }
      throw error;
    }
  }

  async abortUpload(uploadId: string, uploadToken: string): Promise<void> {
    // 验证 Token + 状态（仅 initiated/uploading 可中止）
    const session = await this.validateUploadToken(uploadId, uploadToken, [
      UploadStatus.INITIATED,
      UploadStatus.UPLOADING,
    ]);

    await this.uploadsRepository.update(uploadId, {
      status: UploadStatus.ABORTED,
    });

    await this.storageService.deleteUploadTempDir(uploadId);
  }

  async resumeUpload(uploadId: string, uploadToken: string): Promise<{ received_parts: number[]; chunk_size: number }> {
    // 验证 Token + 状态（仅 initiated/uploading 可恢复）
    const session = await this.validateUploadToken(uploadId, uploadToken, [
      UploadStatus.INITIATED,
      UploadStatus.UPLOADING,
    ]);

    const parts = await this.partsRepository.find({
      where: { uploadId },
      order: { partNumber: 'ASC' },
    });

    return {
      received_parts: parts.map((p) => p.partNumber),
      chunk_size: session.chunkSize,
    };
  }

  private async validateUploadToken(uploadId: string, uploadToken: string, allowedStatuses?: UploadStatus[]): Promise<UploadSession> {
    const tokenHash = this.hashToken(uploadToken);
    const session = await this.uploadsRepository.findOne({
      where: { id: uploadId, uploadTokenHash: tokenHash },
    });

    if (!session) {
      throw new NotFoundException('Upload session not found');
    }

    if (session.expiresAt < Date.now()) {
      throw new GoneException({
        code: 'UPLOAD_EXPIRED',
        message: 'Upload session has expired',
      });
    }

    // 验证状态（如提供）
    if (allowedStatuses && !allowedStatuses.includes(session.status)) {
      throw new BadRequestException({
        code: 'INVALID_UPLOAD_STATE',
        message: `Operation not allowed in state: ${session.status}`,
        allowed: allowedStatuses,
        current: session.status,
      });
    }

    return session;
  }

  private hashToken(token: string): string {
    return createHash('sha256').update(token).digest('hex');
  }
}
```

```typescript
  @Get(':id/content')
  async getContent(
    @Param('id') id: string,
    @Headers('range') rangeHeader: string,
    @Res() res: Response,
  ): Promise<void> {
    // 解析 Range
    let start: number | undefined;
    let end: number | undefined;

    if (rangeHeader) {
      const match = rangeHeader.match(/bytes=(\d*)-(\d*)/);
      if (match) {
        if (match[1]) start = parseInt(match[1], 10);
        if (match[2]) end = parseInt(match[2], 10);
      }
    }

    const { stream, file } = await this.filesService.getFileContent(id, start, end);

    res.setHeader('Content-Type', file.mimeType || 'application/octet-stream');
    res.setHeader('Content-Disposition', `attachment; filename="${encodeURIComponent(file.filename)}"`);
    res.setHeader('Accept-Ranges', 'bytes');

    if (start !== undefined && end !== undefined) {
      res.status(206);
      res.setHeader('Content-Range', `bytes ${start}-${end}/${file.size}`);
      res.setHeader('Content-Length', end - start + 1);
    } else {
      res.setHeader('Content-Length', file.size);
    }

    stream.pipe(res);
```

```typescript
import { Injectable, NotFoundException, BadRequestException, GoneException, ForbiddenException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, DataSource } from 'typeorm';
import { Share, ShareType, ShareProtection } from './entities/share.entity';
import { DownloadSession } from './entities/download-session.entity';
import { File, FileStatus } from '../files/entities/file.entity';
import { v4 as uuidv4 } from 'uuid';
import { createHash, randomBytes } from 'crypto';
import * as bcrypt from 'bcrypt';
import { ShareInfo } from '@filestation/shared';

@Injectable()
export class SharesService {
  constructor(
    @InjectRepository(Share)
    private sharesRepository: Repository<Share>,
    @InjectRepository(DownloadSession)
    private downloadSessionsRepository: Repository<DownloadSession>,
    @InjectRepository(File)
    private filesRepository: Repository<File>,
    private dataSource: DataSource,
  ) {}

  async createShare(
    fileId: string,
    type: ShareType,
    protection: ShareProtection,
    password: string | null,
    maxDownloads: number | null,
    expiresAt: Date | null,
    createdBy: string,
  ): Promise<Share> {
    // 验证文件存在
    const file = await this.filesRepository.findOne({ where: { id: fileId, status: FileStatus.ACTIVE } });
    if (!file) {
      throw new NotFoundException('File not found');
    }

    // Phase 1 禁止 direct 类型
    if (type !== ShareType.PAGE) {
      throw new BadRequestException({
        code: 'INVALID_SHARE_TYPE',
        message: 'Only page shares are supported in Phase 1',
      });
    }

    // 生成分享 ID（短随机码）
    const shareId = this.generateShareId();

    const share = this.sharesRepository.create({
      id: shareId,
      fileId,
      type,
      protection,
      passwordHash: password ? await bcrypt.hash(password, 10) : null,
      maxDownloads,
      usedDownloads: 0,
      status: 'active',
      createdBy,
      createdAt: Date.now(),
      expiresAt: expiresAt ? expiresAt.getTime() : null,
    });

    return this.sharesRepository.save(share);
  }

  async getShareInfo(shareId: string): Promise<ShareInfo> {
    const share = await this.sharesRepository.findOne({
      where: { id: shareId },
      relations: ['file'],
    });

    if (!share) {
      throw new NotFoundException('Share not found');
    }

    // 检查是否过期
    if (share.expiresAt && share.expiresAt < Date.now()) {
      throw new GoneException({
        code: 'SHARE_EXPIRED',
        message: 'Share has expired',
      });
    }

    // 检查是否已吊销
    if (share.status === 'revoked') {
      throw new GoneException({
        code: 'SHARE_REVOKED',
        message: 'Share has been revoked',
      });
    }

    // 检查下载次数
    if (share.maxDownloads !== null && share.usedDownloads >= share.maxDownloads) {
      throw new GoneException({
        code: 'SHARE_EXHAUSTED',
        message: 'Share download limit reached',
      });
    }

    // 检查文件状态
    if (share.file.status !== FileStatus.ACTIVE) {
      throw new GoneException({
        code: 'FILE_UNAVAILABLE',
        message: 'File is no longer available',
      });
    }

    return {
      id: share.id,
      file_id: share.fileId,
      filename: share.file.filename,
      size: share.file.size,
      type: share.type,
      protection: share.protection,
      requires_password: share.protection === ShareProtection.PASSWORD,
      expires_at: share.expiresAt ? new Date(share.expiresAt).toISOString() : null,
      max_downloads: share.maxDownloads,
      used_downloads: share.usedDownloads,
    };
  }

  async verifyPassword(shareId: string, password: string): Promise<{ download_token: string; expires_at: string; file_info: ShareInfo }> {
    // 先验证分享可访问性
    const shareInfo = await this.getShareInfo(shareId);

    if (shareInfo.protection !== ShareProtection.PASSWORD) {
      throw new BadRequestException('Share does not require password');
    }

    // 查询分享获取密码哈希
    const share = await this.sharesRepository.findOne({
      where: { id: shareId },
      relations: ['file'],
    });

    if (!share || !share.passwordHash) {
      throw new BadRequestException('Share password not set');
    }

    const isValid = await bcrypt.compare(password, share.passwordHash);
    if (!isValid) {
      throw new BadRequestException({
        code: 'INVALID_PASSWORD',
        message: 'Invalid password',
      });
    }

    // 创建下载会话
    const downloadToken = uuidv4();
    const expiresAt = Date.now() + 60 * 60 * 1000; // 1 hour

    const session = this.downloadSessionsRepository.create({
      id: uuidv4(),
      shareId: share.id,
      fileId: share.fileId,
      tokenHash: this.hashToken(downloadToken),
      counted: 0,
      createdAt: Date.now(),
      expiresAt,
    });
    await this.downloadSessionsRepository.save(session);

    return {
      download_token: downloadToken,
      expires_at: new Date(expiresAt).toISOString(),
      file_info: shareInfo,
    };
  }

  async createAccess(shareId: string): Promise<{ download_token: string; expires_at: string; file_info: ShareInfo }> {
    // 获取分享信息（验证状态）
    const shareInfo = await this.getShareInfo(shareId);

    // 关键：免密分享才能直接 access
    if (shareInfo.protection !== ShareProtection.NONE) {
      throw new ForbiddenException({
        code: 'VERIFICATION_REQUIRED',
        message: 'Password or admin verification required',
      });
    }

    const downloadToken = uuidv4();
    const expiresAt = Date.now() + 60 * 60 * 1000; // 1 hour

    const session = this.downloadSessionsRepository.create({
      id: uuidv4(),
      shareId,
      fileId: shareInfo.file_id,
      tokenHash: this.hashToken(downloadToken),
      counted: 0,
      createdAt: Date.now(),
      expiresAt,
    });
    await this.downloadSessionsRepository.save(session);

    return {
      download_token: downloadToken,
      expires_at: new Date(expiresAt).toISOString(),
      file_info: shareInfo,
    };
  }

  async validateDownloadToken(token: string): Promise<DownloadSession> {
    const tokenHash = this.hashToken(token);
    const session = await this.downloadSessionsRepository.findOne({
      where: { tokenHash },
      relations: ['share', 'file'],
    });

    if (!session) {
      throw new NotFoundException('Download session not found');
    }

    if (session.expiresAt < Date.now()) {
      throw new GoneException({
        code: 'TOKEN_EXPIRED',
        message: 'Download token has expired',
      });
    }

    // 重新验证分享状态（防吊销后继续使用）
    if (session.share.status === 'revoked') {
      throw new GoneException({
        code: 'SHARE_REVOKED',
        message: 'Share has been revoked',
      });
    }

    if (session.share.expiresAt && session.share.expiresAt < Date.now()) {
      throw new GoneException({
        code: 'SHARE_EXPIRED',
        message: 'Share has expired',
      });
    }

    // 检查下载次数（仅当会话未计数时）
    if (session.counted === 0 && session.share.maxDownloads !== null && session.share.usedDownloads >= session.share.maxDownloads) {
      throw new GoneException({
        code: 'SHARE_EXHAUSTED',
        message: 'Share download limit reached',
      });
    }

    if (session.file.status !== FileStatus.ACTIVE) {
      throw new GoneException({
        code: 'FILE_UNAVAILABLE',
        message: 'File is no longer available',
      });
    }

    return session;
  }

  async revokeShare(shareId: string): Promise<void> {
    const result = await this.sharesRepository.update(
      { id: shareId, status: 'active' },
      { status: 'revoked', revokedAt: Date.now() },
    );

    if (result.affected === 0) {
      throw new NotFoundException('Share not found or already revoked');
    }
  }

  private generateShareId(): string {
    // 使用 crypto.randomBytes 生成 128 bit 随机值
    return randomBytes(16).toString('base64url');
  }

  private hashToken(token: string): string {
    return createHash('sha256').update(token).digest('hex');
  }
}
```

```typescript
import { Injectable, NotFoundException, GoneException, BadRequestException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, DataSource } from 'typeorm';
import { DownloadSession } from './entities/download-session.entity';
import { Share } from './entities/share.entity';
import { File, FileStatus } from '../files/entities/file.entity';
import { StorageService } from '../files/storage.service';

export interface DownloadResult {
  stream: NodeJS.ReadableStream;
  file: File;
  session: DownloadSession;
  start: number;
  end: number;
  totalSize: number;
  isPartial: boolean;
}

@Injectable()
export class DownloadService {
  constructor(
    @InjectRepository(DownloadSession)
    private downloadSessionsRepository: Repository<DownloadSession>,
    @InjectRepository(Share)
    private sharesRepository: Repository<Share>,
    @InjectRepository(File)
    private filesRepository: Repository<File>,
    private storageService: StorageService,
    private dataSource: DataSource,
  ) {}

  async prepareDownload(
    session: DownloadSession,
    rangeHeader?: string,
  ): Promise<DownloadResult> {
    // 解析 Range（统一 RangeParser）
    const range = this.parseRange(rangeHeader, session.file.size);
    if (!range.valid) {
      throw new BadRequestException({
        code: 'INVALID_RANGE',
        message: 'Invalid range',
        details: { start: range.start, end: range.end, size: session.file.size },
      });
    }

    // 先抢占计数权（在创建流之前）
    await this.countDownload(session);

    // 获取文件流（真实 Range）
    const stream = await this.storageService.getFileStream(session.file.storedName, range.start, range.end);

    return {
      stream,
      file: session.file,
      session,
      start: range.start,
      end: range.end,
      totalSize: session.file.size,
      isPartial: range.isPartial,
    };
  }

  private parseRange(rangeHeader: string | undefined, fileSize: number): { valid: boolean; start: number; end: number; isPartial: boolean } {
    if (!rangeHeader) {
      return { valid: true, start: 0, end: fileSize - 1, isPartial: false };
    }

    const match = rangeHeader.match(/^bytes=(\d*)-(\d*)$/);
    if (!match) {
      return { valid: false, start: 0, end: 0, isPartial: false };
    }

    let start: number;
    let end: number;

    if (match[1] && match[2]) {
      // bytes=0-499
      start = parseInt(match[1], 10);
      end = parseInt(match[2], 10);
    } else if (match[1] && !match[2]) {
      // bytes=500-
      start = parseInt(match[1], 10);
      end = fileSize - 1;
    } else if (!match[1] && match[2]) {
      // bytes=-500 （最后 500 字节）
      const suffixLength = parseInt(match[2], 10);
      start = Math.max(0, fileSize - suffixLength);
      end = fileSize - 1;
    } else {
      return { valid: false, start: 0, end: 0, isPartial: false };
    }

    // 验证范围
    if (start > end || start >= fileSize || end >= fileSize) {
      return { valid: false, start, end, isPartial: false };
    }

    return { valid: true, start, end, isPartial: true };
  }

  private async countDownload(session: DownloadSession): Promise<void> {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    await queryRunner.startTransaction();

    try {
      // 1. 先抢占 download_session 计数权（原子）
      const sessionResult = await queryRunner.manager
        .createQueryBuilder()
        .update(DownloadSession)
        .set({ counted: 1, countedAt: Date.now() })
        .where('id = :id', { id: session.id })
        .andWhere('counted = 0')
        .andWhere('expires_at > :now', { now: Date.now() })
        .execute();

      if (sessionResult.affected === 0) {
        // 已计数，直接返回
        await queryRunner.commitTransaction();
        return;
      }

      // 2. 扣减分享额度（原子）
      const shareResult = await queryRunner.manager
        .createQueryBuilder()
        .update(Share)
        .set({
          usedDownloads: () => 'used_downloads + 1',
          lastUsedAt: Date.now(),
        })
        .where('id = :id', { id: session.shareId })
        .andWhere('status = :status', { status: 'active' })
        .andWhere('(expires_at IS NULL OR expires_at > :now)', { now: Date.now() })
        .andWhere('(max_downloads IS NULL OR used_downloads < max_downloads)')
        .execute();

      if (shareResult.affected === 0) {
        throw new GoneException({
          code: 'SHARE_EXHAUSTED',
          message: 'Share download limit reached',
        });
      }

      await queryRunner.commitTransaction();
    } catch (error) {
      await queryRunner.rollbackTransaction();
      throw error;
    } finally {
      await queryRunner.release();
    }
  }
}
```

```typescript
import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, Not } from 'typeorm';
import { Folder } from './entities/folder.entity';
import { File, FileStatus } from '../files/entities/file.entity';
import { v4 as uuidv4 } from 'uuid';

@Injectable()
export class FoldersService {
  constructor(
    @InjectRepository(Folder)
    private foldersRepository: Repository<Folder>,
    @InjectRepository(File)
    private filesRepository: Repository<File>,
  ) {}

  async create(name: string, parentId: string | null, createdBy: string): Promise<Folder> {
    // 检查父文件夹存在
    if (parentId) {
      const parent = await this.foldersRepository.findOne({ where: { id: parentId, isDeleted: 0 } });
      if (!parent) {
        throw new NotFoundException('Parent folder not found');
      }
    }

    const folder = this.foldersRepository.create({
      id: uuidv4(),
      name,
      parentId,
      createdBy,
      createdAt: Date.now(),
      isDeleted: 0,
    });

    return this.foldersRepository.save(folder);
  }

  async findAll(): Promise<Folder[]> {
    return this.foldersRepository.find({
      where: { isDeleted: 0 },
      order: { name: 'ASC' },
    });
  }

  async findTree(): Promise<Folder[]> {
    const allFolders = await this.foldersRepository.find({
      where: { isDeleted: 0 },
      order: { name: 'ASC' },
    });

    // 组装树
    const folderMap = new Map<string, Folder & { children: Folder[] }>();
    const roots: (Folder & { children: Folder[] })[] = [];

    // 初始化映射
    for (const folder of allFolders) {
      folderMap.set(folder.id, { ...folder, children: [] });
    }

    // 组装树
    for (const folder of allFolders) {
      const node = folderMap.get(folder.id)!;
      if (folder.parentId && folderMap.has(folder.parentId)) {
        folderMap.get(folder.parentId)!.children.push(node);
      } else {
        roots.push(node);
      }
    }

    return roots;
  }

  async findOne(id: string): Promise<Folder> {
    const folder = await this.foldersRepository.findOne({ where: { id, isDeleted: 0 } });
    if (!folder) {
      throw new NotFoundException('Folder not found');
    }
    return folder;
  }

  async update(id: string, name: string): Promise<Folder> {
    const folder = await this.findOne(id);
    folder.name = name;
    return this.foldersRepository.save(folder);
  }

  async move(id: string, newParentId: string | null): Promise<Folder> {
    const folder = await this.findOne(id);

    // 防止循环移动
    if (newParentId) {
      if (newParentId === id) {
        throw new BadRequestException('Cannot move folder to itself');
      }

      // 检查是否移动到后代
      const descendants = await this.getDescendantIds(id);
      if (descendants.includes(newParentId)) {
        throw new BadRequestException('Cannot move folder to its descendant');
      }
    }

    folder.parentId = newParentId;
    return this.foldersRepository.save(folder);
  }

  async delete(id: string): Promise<void> {
    const folder = await this.findOne(id);

    // 检查是否有子文件夹
    const childCount = await this.foldersRepository.count({
      where: { parentId: id, isDeleted: 0 },
    });
    if (childCount > 0) {
      throw new BadRequestException({
        code: 'FOLDER_NOT_EMPTY',
        message: 'Cannot delete folder with subfolders',
      });
    }

    // 检查是否有文件（所有非 deleted 状态）
    const fileCount = await this.filesRepository.count({
      where: { folderId: id, status: Not(FileStatus.DELETED) },
    });
    if (fileCount > 0) {
      throw new BadRequestException({
        code: 'FOLDER_NOT_EMPTY',
        message: 'Cannot delete folder with files',
      });
    }

    folder.isDeleted = 1;
    folder.deletedAt = Date.now();
    await this.foldersRepository.save(folder);
  }

  private async getDescendantIds(folderId: string): Promise<string[]> {
    const result: string[] = [];
    const queue = [folderId];

    while (queue.length > 0) {
      const currentId = queue.shift()!;
      const children = await this.foldersRepository.find({
        where: { parentId: currentId, isDeleted: 0 },
      });

      for (const child of children) {
        result.push(child.id);
        queue.push(child.id);
      }
    }

    return result;
  }
}
```

```typescript
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { ApiResponse } from '@filestation/shared';
import { Request } from 'express';

@Controller('settings')
export class SettingsController {
  constructor(private settingsService: SettingsService) {}

  @Get()
  @UseGuards(JwtAuthGuard)
  async getAll(): Promise<ApiResponse<any>> {
    const [site, security, transfer, storage] = await Promise.all([
      this.settingsService.getSiteSettings(),
      this.settingsService.getSecuritySettings(),
      this.settingsService.getTransferSettings(),
      this.settingsService.getStorageSettings(),
    ]);

    return {
      code: 'OK',
      message: 'Settings retrieved',
      data: { site, security, transfer, storage },
      request_id: crypto.randomUUID(),
    };
  }

  @Get('public')
  async getPublic(): Promise<ApiResponse<{ name: string; icon: string | null }>> {
    const site = await this.settingsService.getSiteSettings();
    return {
      code: 'OK',
      message: 'Public settings retrieved',
      data: { name: site.name, icon: site.icon },
      request_id: crypto.randomUUID(),
    };
  }

  @Put()
  @UseGuards(JwtAuthGuard)
  async update(
    @Body() body: { site?: any; security?: any; transfer?: any; storage?: any },
    @Req() req: Request,
  ): Promise<ApiResponse<null>> {
    if (body.site) await this.settingsService.set('site', body.site, (req as any).user.id);
    if (body.security) await this.settingsService.set('security', body.security, (req as any).user.id);
    if (body.transfer) await this.settingsService.set('transfer', body.transfer, (req as any).user.id);
    
    // 处理 storage：剔除不可修改的 path 字段
    if (body.storage) {
      const { path: _ignored, ...safeStorage } = body.storage;
      if (Object.keys(safeStorage).length > 0) {
        await this.settingsService.set('storage', safeStorage, (req as any).user.id);
      }
    }

    return {
      code: 'OK',
      message: 'Settings updated',
      data: null,
      request_id: crypto.randomUUID(),
    };
  }
```

```typescript
import { ApiResponse } from '@filestation/shared';

const API_BASE = '/api/v1';

class ApiClient {
  private accessToken: string | null = null;

  setAccessToken(token: string | null) {
    this.accessToken = token;
  }

  getAccessToken(): string | null {
    return this.accessToken;
  }

  private async request<T>(
    method: string,
    path: string,
    body?: BodyInit | object,
    headers: Record<string, string> = {},
  ): Promise<ApiResponse<T>> {
    const url = `${API_BASE}${path}`;
    const requestHeaders: Record<string, string> = { ...headers };

    // 判断是否为二进制数据
    const isRaw = body instanceof Blob || body instanceof ArrayBuffer || body instanceof FormData;

    if (!isRaw && body !== undefined) {
      requestHeaders['Content-Type'] = 'application/json';
    }

    if (this.accessToken) {
      requestHeaders['Authorization'] = `Bearer ${this.accessToken}`;
    }

    const response = await fetch(url, {
      method,
      headers: requestHeaders,
      body: isRaw ? (body as BodyInit) : body === undefined ? undefined : JSON.stringify(body),
      credentials: 'include', // 包含 HttpOnly Cookie
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({ code: 'UNKNOWN', message: 'Unknown error' }));
      throw new Error(error.message || 'Request failed');
    }

    return response.json();
  }

  async get<T>(path: string): Promise<ApiResponse<T>> {
    return this.request<T>('GET', path);
  }

  async post<T>(path: string, body?: BodyInit | object, headers?: Record<string, string>): Promise<ApiResponse<T>> {
    return this.request<T>('POST', path, body, headers);
  }

  async put<T>(path: string, body?: BodyInit | object, headers?: Record<string, string>): Promise<ApiResponse<T>> {
    return this.request<T>('PUT', path, body, headers);
  }

  async patch<T>(path: string, body?: unknown): Promise<ApiResponse<T>> {
    return this.request<T>('PATCH', path, body);
  }

  async delete<T>(path: string): Promise<ApiResponse<T>> {
    return this.request<T>('DELETE', path);
  }
}

export const api = new ApiClient();
```

```typescript
import { create } from 'zustand';
import { api } from '../lib/api';

interface AuthState {
  accessToken: string | null;
  isAuthenticated: boolean;
  username: string | null;
  login: (accessToken: string, username: string) => void;
  logout: () => void;
  setAccessToken: (token: string) => void;
  checkAuth: () => Promise<boolean>;
}

export const useAuthStore = create<AuthState>((set, get) => ({
  accessToken: null,
  isAuthenticated: false,
  username: null,

  login: (accessToken, username) => {
    api.setAccessToken(accessToken);
    set({ accessToken, isAuthenticated: true, username });
  },

  logout: async () => {
    // 调用后端 logout 吊销会话并清除 Cookie
    try {
      await api.post('/auth/logout');
    } catch (err) {
      console.error('Logout failed:', err);
    }
    api.setAccessToken(null);
    set({ accessToken: null, isAuthenticated: false, username: null });
  },

  setAccessToken: (token) => {
    api.setAccessToken(token);
    set({ accessToken: token });
  },

  checkAuth: (() => {
    let promise: Promise<boolean> | null = null;
    
    return async () => {
      // Single-flight：并发调用复用同一 Promise
      if (promise) {
        return promise;
      }

      promise = (async () => {
        try {
          const response = await api.post<{ access_token: string; expires_in: number; username: string }>('/auth/refresh');
          const { access_token, username } = response.data!;
          api.setAccessToken(access_token);
          set({ accessToken: access_token, isAuthenticated: true, username });
          return true;
        } catch {
          return false;
        } finally {
          promise = null;
        }
      })();

      return promise;
    };
  })(),
}));
```

```tsx
import { useState, useCallback } from 'react';
import { useDropzone } from 'react-dropzone';
import { api } from '../lib/api';
import { UploadInitResponse } from '@filestation/shared';

interface FileUploadProps {
  onUploadComplete: () => void;
}

export default function FileUpload({ onUploadComplete }: FileUploadProps) {
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);

  const onDrop = useCallback(async (acceptedFiles: File[]) => {
    if (acceptedFiles.length === 0) return;

    const file = acceptedFiles[0];
    setUploading(true);
    setProgress(0);

    try {
      // 1. 初始化上传
      const initResponse = await api.post<UploadInitResponse>('/uploads', {
        filename: file.name,
        size: file.size,
      });

      const { upload_id, upload_token, chunk_size } = initResponse.data!;

      // 2. 分块上传
      const totalChunks = Math.ceil(file.size / chunk_size);
      for (let i = 0; i < totalChunks; i++) {
        const start = i * chunk_size;
        const end = Math.min(start + chunk_size, file.size);
        const chunk = file.slice(start, end);

        // 计算 checksum
        const buffer = await chunk.arrayBuffer();
        const hashBuffer = await crypto.subtle.digest('SHA-256', buffer);
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        const checksum = hashArray.map((b) => b.toString(16).padStart(2, '0')).join('');

        await api.put(`/uploads/${upload_id}/parts/${i}`, chunk, {
          'X-Upload-Token': upload_token,
          'X-Part-Checksum': checksum,
        });

        setProgress(Math.round(((i + 1) / totalChunks) * 100));
      }

      // 3. 完成上传
      await api.post(`/uploads/${upload_id}/complete`, {}, {
        'X-Upload-Token': upload_token,
      });

      onUploadComplete();
    } catch (err) {
      console.error('Upload failed:', err);
      alert('上传失败');
    } finally {
      setUploading(false);
      setProgress(0);
    }
  }, [onUploadComplete]);

  const { getRootProps, getInputProps, isDragActive } = useDropzone({ onDrop });

  return (
    <div
      {...getRootProps()}
      className={`border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors ${
        isDragActive ? 'border-blue-500 bg-blue-50' : 'border-gray-300 hover:border-gray-400'
      }`}
    >
      <input {...getInputProps()} />
      {uploading ? (
        <div>
          <div className="text-lg mb-2">上传中... {progress}%</div>
          <div className="w-full bg-gray-200 rounded-full h-2">
            <div
              className="bg-blue-600 h-2 rounded-full transition-all"
              style={{ width: `${progress}%` }}
            />
          </div>
        </div>
      ) : (
        <div>
          <p className="text-lg">拖拽文件到此处，或点击选择文件</p>
          <p className="text-sm text-gray-500 mt-2">支持大文件分块上传</p>
        </div>
      )}
    </div>
  );
}
```
